function normalizeProvider(
  provider: string,
): string {
  return provider
    .trim()
    .toLowerCase();
}

function normalizeModel(
  model: string,
): string {
  return model.trim();
}

export function createModelRegistryId(
  provider: string,
  model: string,
): string {
  const normalizedProvider =
    normalizeProvider(provider);

  const normalizedModel =
    normalizeModel(model);

  if (!normalizedProvider) {
    throw new Error(
      "Model provider must not be empty.",
    );
  }

  if (!normalizedModel) {
    throw new Error(
      "Model name must not be empty.",
    );
  }

  return `${normalizedProvider}:${normalizedModel}`;
}

function cloneEntry(
  entry: ModelRegistryEntry,
): ModelRegistryEntry {
  return structuredClone(entry);
}

function uniqueSorted<T extends string>(
  values: readonly T[],
): T[] {
  return [
    ...new Set(values),
  ].sort() as T[];
}

function normalizeEntry(
  entry: ModelRegistryEntry,
): ModelRegistryEntry {
  const provider =
    normalizeProvider(entry.provider);

  const model =
    normalizeModel(entry.model);

  const id =
    createModelRegistryId(
      provider,
      model,
    );

  if (
    entry.id &&
    entry.id !== id
  ) {
    throw new Error(
      `Model registry id mismatch: expected ${id}, received ${entry.id}`,
    );
  }

  return {
    ...structuredClone(entry),
    id,
    provider,
    model,
    displayName:
      entry.displayName.trim() ||
      model,
    roles:
      uniqueSorted(entry.roles),
    capabilities:
      uniqueSorted(
        entry.capabilities,
      ),
    version:
      entry.version?.trim() ||
      undefined,
    strengths:
      uniqueSorted(
        entry.strengths
          .map(
            (value) =>
              value.trim(),
          )
          .filter(Boolean),
      ),
    weaknesses:
      uniqueSorted(
        entry.weaknesses
          .map(
            (value) =>
              value.trim(),
          )
          .filter(Boolean),
      ),
    tags:
      uniqueSorted(
        entry.tags
          .map(
            (tag) =>
              tag.trim(),
          )
          .filter(Boolean),
      ),
  };
}

export class ModelRegistry {
  private readonly entries =
    new Map<
      string,
      ModelRegistryEntry
    >();

  register(
    entry: ModelRegistryEntry,
  ): ModelRegistryEntry {
    const normalized =
      normalizeEntry(entry);

    if (
      this.entries.has(
        normalized.id,
      )
    ) {
      throw new Error(
        `Model is already registered: ${normalized.id}`,
      );
    }

    this.entries.set(
      normalized.id,
      normalized,
    );

    return cloneEntry(
      normalized,
    );
  }

  upsert(
    entry: ModelRegistryEntry,
  ): ModelRegistryEntry {
    const normalized =
      normalizeEntry(entry);

    this.entries.set(
      normalized.id,
      normalized,
    );

    return cloneEntry(
      normalized,
    );
  }

  get(
    id: string,
  ): ModelRegistryEntry | undefined {
    const entry =
      this.entries.get(id);

    return entry
      ? cloneEntry(entry)
      : undefined;
  }

  getByProviderModel(
    provider: string,
    model: string,
  ): ModelRegistryEntry | undefined {
    return this.get(
      createModelRegistryId(
        provider,
        model,
      ),
    );
  }

  has(
    id: string,
  ): boolean {
    return this.entries.has(id);
  }

  remove(
    id: string,
  ): boolean {
    return this.entries.delete(id);
  }

  list(
    query: ModelRegistryQuery = {},
  ): ModelRegistryEntry[] {
    const provider =
      query.provider
        ? normalizeProvider(
            query.provider,
          )
        : undefined;

    return [
      ...this.entries.values(),
    ]
      .filter(
        (entry) =>
          (
            provider === undefined ||
            entry.provider === provider
          ) &&
          (
            query.role === undefined ||
            entry.roles.includes(
              query.role,
            )
          ) &&
          (
            query.capability === undefined ||
            entry.capabilities.includes(
              query.capability,
            )
          ) &&
          (
            query.enabled === undefined ||
            entry.enabled ===
              query.enabled
          ) &&
          (
            query.local === undefined ||
            entry.local ===
              query.local
          ),
      )
      .sort(
        (a, b) =>
          a.id.localeCompare(
            b.id,
          ),
      )
      .map(cloneEntry);
  }
}

function capabilitiesForRole(
  role: ModelRole,
): ModelCapability[] {
  switch (role) {
    case "planner":
      return [
        "planning",
      ];

    case "code":
      return [
        "coding",
      ];

    case "repair":
      return [
        "repair",
      ];

    case "default":
    default:
      return [
        "chat",
      ];
  }
}

export function createConfiguredModelRegistry():
  ModelRegistry {
  const registry =
    new ModelRegistry();

  const roles:
    ModelRole[] = [
      "default",
      "code",
      "planner",
      "repair",
    ];

  for (const role of roles) {
    const candidates =
      getModelCandidates(role);

    for (
      const candidate
      of candidates
    ) {
      const id =
        createModelRegistryId(
          "ollama",
          candidate.model,
        );

      const existing =
        registry.get(id);

      if (existing) {
        registry.upsert({
          ...existing,
          roles:
            uniqueSorted([
              ...existing.roles,
              role,
            ]),
          capabilities:
            uniqueSorted([
              ...existing.capabilities,
              ...capabilitiesForRole(
                role,
              ),
            ]),
          tags:
            uniqueSorted([
              ...existing.tags,
              candidate.source,
            ]),
        });

        continue;
      }

      registry.register({
        id,
        provider:
          "ollama",
        model:
          candidate.model,
        displayName:
          candidate.model,
        roles: [
          role,
        ],
        capabilities:
          capabilitiesForRole(
            role,
          ),
        enabled:
          true,
        local:
          true,
        availability:
          "unknown",
        contextLimits: {
          known:
            false,
        },
        cost: {
          known:
            true,
          mode:
            "local-compute",
        },
        privacy: {
          execution:
            "local",
          promptLeavesDevice:
            false,
          retention:
            "unknown",
        },
        strengths: [
          ...capabilitiesForRole(
            role,
          ),
        ],
        weaknesses: [
          "Empirical performance not yet calibrated",
        ],
        source:
          "configuration",
        tags: [
          candidate.source,
        ],
      });
    }
  }

  return registry;
}

const defaultModelRegistry =

export function getDefaultModelRegistry():

  return defaultModelRegistry;

function average(
  values:
    readonly number[],
): number | null {
  if (
    values.length === 0
  ) {
    return null;
  }

  return (
    values.reduce(
      (sum, value) =>
        sum + value,
      0,
    ) /
    values.length
  );
}

function buildScore(
  subjectKind:
    CalibrationCapabilityScore["subjectKind"],
  subjectId:
    string,
  taskClass:
    string,
  samples:
    readonly CalibrationSampleResult[],
  blind:
    boolean,
): CalibrationCapabilityScore {
  const executions =
    samples.filter(
      (sample) =>
        sample.execution.ok,
    );

  const evaluated =
    samples.filter(
      (sample) =>
        sample.evaluation,
    );

  const objectiveScores =
    evaluated.flatMap(
      (sample) => {
        const score =
          sample.evaluation
            ?.objective.score;

        return typeof score ===
          "number"
          ? [
              score,
            ]
          : [];
      },
    );

  const subjectiveScores =
    evaluated.flatMap(
      (sample) => {
        const score =
          sample.evaluation
            ?.subjective.score;

        return typeof score ===
          "number"
          ? [
              score,
            ]
          : [];
      },
    );

  const passes =
    evaluated.filter(
      (sample) =>
        sample.evaluation
          ?.outcome ===
        "pass",
    ).length;

  return {
    subjectKind,
    subjectId,
    taskClass,
    sampleCount:
      samples.length,
    executionSuccessRate:
      samples.length > 0
        ? executions.length /
          samples.length
        : 0,
    passRate:
      samples.length > 0
        ? passes /
          samples.length
        : 0,
    objectiveScore:
      average(
        objectiveScores,
      ),
    subjectiveScore:
      average(
        subjectiveScores,
      ),
    blindSamples:
      blind
        ? samples.length
        : 0,
  };
}

function groupScores(
  samples:
    readonly CalibrationSampleResult[],
  candidates:
    readonly CalibrationCandidate[],
  blind:
    boolean,
): CalibrationCapabilityScore[] {
  const scores:
    CalibrationCapabilityScore[] =
      [];

  const taskClasses =
    [
      ...new Set(
        samples.map(
          (sample) =>
            sample.taskClass,
        ),
      ),
    ].sort();

  for (
    const taskClass
    of taskClasses
  ) {
    for (
      const candidate
      of candidates
    ) {
      const relevant =
        samples.filter(
          (sample) =>
            sample.taskClass ===
              taskClass &&
            sample.candidateId ===
              candidate.id,
        );

      if (
        relevant.length === 0
      ) {
        continue;
      }

      scores.push(
        buildScore(
          "candidate",
          candidate.id,
          taskClass,
          relevant,
          blind,
        ),
      );
    }

    const modelIds =
      [
        ...new Set(
          candidates.map(
            (candidate) =>
              candidate.modelId,
          ),
        ),
      ].sort();

    for (
      const modelId
      of modelIds
    ) {
      const candidateIds =
        new Set(
          candidates
            .filter(
              (candidate) =>
                candidate.modelId ===
                modelId,
            )
            .map(
              (candidate) =>
                candidate.id,
            ),
        );

      const relevant =
        samples.filter(
          (sample) =>
            sample.taskClass ===
              taskClass &&
            candidateIds.has(
              sample.candidateId,
            ),
        );

      if (
        relevant.length === 0
      ) {
        continue;
      }

      scores.push(
        buildScore(
          "model",
          modelId,
          taskClass,
          relevant,
          blind,
        ),
      );
    }

    const strategyIds =
      [
        ...new Set(
          candidates.map(
            (candidate) =>
              candidate.strategyId,
          ),
        ),
      ].sort();

    for (
      const strategyId
      of strategyIds
    ) {
      const candidateIds =
        new Set(
          candidates
            .filter(
              (candidate) =>
                candidate.strategyId ===
                strategyId,
            )
            .map(
              (candidate) =>
                candidate.id,
            ),
        );

      const relevant =
        samples.filter(
          (sample) =>
            sample.taskClass ===
              taskClass &&
            candidateIds.has(
              sample.candidateId,
            ),
        );

      if (
        relevant.length === 0
      ) {
        continue;
      }

      scores.push(
        buildScore(
          "strategy",
          strategyId,
          taskClass,
          relevant,
          blind,
        ),
      );
    }
  }

  return scores.sort(
    (left, right) =>
      [
        left.taskClass,
        left.subjectKind,
        left.subjectId,
      ]
        .join(":")
        .localeCompare(
          [
            right.taskClass,
            right.subjectKind,
            right.subjectId,
          ].join(":"),
        ),
  );
}

export class CalibrationArena {
  constructor(
    private readonly evaluation:
      EvaluationFramework,
    private readonly executor:
      CalibrationExecutor,
  ) {}

  async run(
    candidates:
      readonly CalibrationCandidate[],
    tasks:
      readonly CalibrationTask[],
    options:
      CalibrationArenaRunOptions = {},
  ): Promise<CalibrationArenaResult> {
    if (
      candidates.length < 2
    ) {
      throw new Error(
        "Calibration requires at least two candidates.",
      );
    }

    if (
      tasks.length === 0
    ) {
      throw new Error(
        "Calibration requires at least one task.",
      );
    }

    const normalizedCandidates =
      candidates.map(
        normalizeCandidate,
      );

    const normalizedTasks =
      tasks.map(
        normalizeTask,
      );

    const candidateIds =
      normalizedCandidates.map(
        (candidate) =>
          candidate.id,
      );

    if (
      new Set(
        candidateIds,
      ).size !==
      candidateIds.length
    ) {
      throw new Error(
        "Calibration candidate ids must be unique.",
      );
    }

    const taskIds =
      normalizedTasks.map(
        (task) =>
          task.id,
      );

    if (
      new Set(
        taskIds,
      ).size !==
      taskIds.length
    ) {
      throw new Error(
        "Calibration task ids must be unique.",
      );
    }

    const blind =
      options.blind !==
      false;

    const blindMap:
      Record<
        string,
        string
      > =
        Object.fromEntries(
          normalizedCandidates.map(
            (
              candidate,
              index,
            ) => [
              candidate.id,
              `candidate-${String(
                index + 1,
              ).padStart(
                2,
                "0",
              )}`,
            ],
          ),
        );

    const startedAt =
      new Date()
        .toISOString();

    const samples:
      CalibrationSampleResult[] =
        [];

    for (
      const task
      of normalizedTasks
    ) {
      for (
        const candidate
        of normalizedCandidates
      ) {
        let execution:
          CalibrationExecution;

        const executionStarted =
          Date.now();

        try {
          const raw =
            await this.executor(
              structuredClone(
                candidate,
              ),
              structuredClone(
                task,
              ),
            );

          execution = {
            ...structuredClone(
              raw,
            ),
            candidateId:
              candidate.id,
            taskId:
              task.id,
            durationMs:
              Number.isFinite(
                raw.durationMs,
              )
                ? Math.max(
                    0,
                    raw.durationMs,
                  )
                : Date.now() -
                  executionStarted,
          };
        } catch (error) {
          execution = {
            candidateId:
              candidate.id,
            taskId:
              task.id,
            ok:
              false,
            error:
              error instanceof Error
                ? error.message
                : String(error),
            durationMs:
              Date.now() -
              executionStarted,
          };
        }

        let evaluation:
          EvaluationReport
          | undefined;

        if (
          execution.ok
        ) {
          evaluation =
            await this.evaluation
              .evaluate({
                taskClass:
                  task.taskClass,
                input:
                  structuredClone(
                    task.input,
                  ),
                output:
                  structuredClone(
                    execution.output,
                  ),
                expected:
                  structuredClone(
                    task.expected,
                  ),
                provider:
                  blind
                    ? undefined
                    : candidate.provider,
                model:
                  blind
                    ? undefined
                    : candidate.modelId,
                metadata: {
                  calibration:
                    true,
                  taskId:
                    task.id,
                  candidateLabel:
                    blind
                      ? blindMap[
                          candidate.id
                        ]
                      : candidate.id,
                  strategyId:
                    blind
                      ? undefined
                      : candidate.strategyId,
                },
              });
        }

        samples.push({
          candidateId:
            candidate.id,
          blindLabel:
            blindMap[
              candidate.id
            ],
          provider:
            candidate.provider,
          modelId:
            candidate.modelId,
          strategyId:
            candidate.strategyId,
          taskId:
            task.id,
          taskClass:
            task.taskClass,
          execution,
          evaluation,
        });
      }
    }

    return {
      startedAt,
      completedAt:
        new Date()
          .toISOString(),
      blind,
      candidates:
        structuredClone(
          normalizedCandidates,
        ),
      tasks:
        structuredClone(
          normalizedTasks,
        ),
      blindMap,
      samples,
      capabilityScores:
        groupScores(
          samples,
          normalizedCandidates,
          blind,
        ),
    };
  }
}

  };

  firstObservedAt: string;
  lastObservedAt: string;

  lastScheduledEvaluationAt?:
    string;

  recalibration: {
    due: boolean;

    reasons: string[];
  };
}

const DAY_MS =
  24 *
  60 *
  60 *
  1000;

export const DEFAULT_CONTINUOUS_CALIBRATION_POLICY:
  ContinuousCalibrationPolicy = {
    halfLifeDays:
      30,

    scheduledEvaluationIntervalDays:
      14,

    minimumEffectiveSamples:
      2,

    strongSampleCount:
      12,

    maximumObservations:
      5000,

    sourceWeights: {
      realWork:
        0.9,

      scheduledEvaluation:
        1,
    },
  };

function requireFinite(
  value: number,
  field: string,
): number {
  if (
    !Number.isFinite(
      value,
    )
  ) {
    throw new Error(
      `${field} must be finite.`,
    );
  }

  return value;
}

function unitInterval(
  value: number,
  field: string,
): number {
  requireFinite(
    value,
    field,
  );

  if (
    value < 0 ||
    value > 1
  ) {
    throw new Error(
      `${field} must be between 0 and 1.`,
    );
  }

  return value;
}

function nonNegative(
  value:
    number
    | undefined,
  field: string,
): number | undefined {
  if (
    value ===
    undefined
  ) {
    return undefined;
  }

  requireFinite(
    value,
    field,
  );

  if (
    value < 0
  ) {
    throw new Error(
      `${field} must be non-negative.`,
    );
  }

  return value;
}

function nonNegativeInteger(
  value:
    number
    | undefined,
  field: string,
): number | undefined {
  if (
    value ===
    undefined
  ) {
    return undefined;
  }

  if (
    !Number.isInteger(
      value,
    ) ||
    value < 0
  ) {
    throw new Error(
      `${field} must be a non-negative integer.`,
    );
  }

  return value;
}

function requireTimestamp(
  value: string,
  field: string,
): string {
  const parsed =
    new Date(
      value,
    );

  if (
    Number.isNaN(
      parsed.getTime(),
    )
  ) {
    throw new Error(
      `${field} must be a valid timestamp.`,
    );
  }

  return parsed.toISOString();
}

function normalizeText(
  value: string,
  field: string,
): string {
  const normalized =
    value.trim();

  if (!normalized) {
    throw new Error(
      `${field} must not be empty.`,
    );
  }

  return normalized;
}

function optionalText(
  value:
    string
    | undefined,
): string | undefined {
  const normalized =
    value?.trim();

  return normalized ||
    undefined;
}

function normalizeDomain(
  value: string,
): string {
  return normalizeText(
    value,
    "calibration.task.domain",
  )
    .toLowerCase()
    .replace(
      /\s+/g,
      "-",
    );
}

function validatePolicy(
  policy:
    ContinuousCalibrationPolicy,
): ContinuousCalibrationPolicy {
  for (
    const [
      field,
      value,
    ]
    of [
      [
        "halfLifeDays",
        policy.halfLifeDays,
      ],
      [
        "scheduledEvaluationIntervalDays",
        policy.scheduledEvaluationIntervalDays,
      ],
      [
        "minimumEffectiveSamples",
        policy.minimumEffectiveSamples,
      ],
      [
        "strongSampleCount",
        policy.strongSampleCount,
      ],
      [
        "maximumObservations",
        policy.maximumObservations,
      ],
    ] as const
  ) {
    if (
      !Number.isFinite(
        value,
      ) ||
      value <= 0
    ) {
      throw new Error(
        `continuousCalibration.${field} must be greater than zero.`,
      );
    }
  }

  unitInterval(
    policy.sourceWeights
      .realWork,
    "continuousCalibration.sourceWeights.realWork",
  );

  unitInterval(
    policy.sourceWeights
      .scheduledEvaluation,
    "continuousCalibration.sourceWeights.scheduledEvaluation",
  );

  return structuredClone(
    policy,
  );
}

function normalizeObservation(
  value:
    CalibrationObservation,
): CalibrationObservation {
  const observedAt =
    requireTimestamp(
      value.observedAt,
      "calibration.observedAt",
    );

  const objectiveScore =
    unitInterval(
      value.objective
        .score,
      "calibration.objective.score",
    );

  const evidenceConfidence =
    value.objective
      .evidenceConfidence ===
      undefined
      ? undefined
      : unitInterval(
          value.objective
            .evidenceConfidence,
          "calibration.objective.evidenceConfidence",
        );

  const preferenceScore =
    value.subjective
      ?.preferenceScore ===
      undefined
      ? undefined
      : unitInterval(
          value.subjective
            .preferenceScore,
          "calibration.subjective.preferenceScore",
        );

  const userRating =
    value.subjective
      ?.userRating ===
      undefined
      ? undefined
      : unitInterval(
          value.subjective
            .userRating,
          "calibration.subjective.userRating",
        );

  return {
    id:
      normalizeText(
        value.id,
        "calibration.id",
      ),

    observedAt,

    source:
      value.source,

    modelId:
      normalizeText(
        value.modelId,
        "calibration.modelId",
      ),

    modelVersion:
      normalizeText(
        value.modelVersion,
        "calibration.modelVersion",
      ),

    strategy:
      optionalText(
        value.strategy,
      ),

    task: {
      domain:
        normalizeDomain(
          value.task.domain,
        ),

      complexity:
        optionalText(
          value.task
            .complexity,
        ),

      creativity:
        optionalText(
          value.task
            .creativity,
        ),

      factuality:
        optionalText(
          value.task
            .factuality,
        ),

      reasoningDepth:
        optionalText(
          value.task
            .reasoningDepth,
        ),

      privacySensitivity:
        optionalText(
          value.task
            .privacySensitivity,
        ),

      toolUse:
        optionalText(
          value.task
            .toolUse,
        ),
    },

    objective: {
      score:
        objectiveScore,

      success:
        value.objective
          .success,

      verificationVerdict:
        value.objective
          .verificationVerdict ??
        "unknown",

      executionSucceeded:
        value.objective
          .executionSucceeded,

      evidenceConfidence,
    },

    subjective: value.subjective
      ? {
          preferenceScore,

          userRating,

          source:
            optionalText(
              value.subjective
                .source,
            ),
        }
      : undefined,

    telemetry: value.telemetry
      ? {
          latencyMs:
            nonNegative(
              value.telemetry
                .latencyMs,
              "calibration.telemetry.latencyMs",
            ),

          cost:
            nonNegative(
              value.telemetry
                .cost,
              "calibration.telemetry.cost",
            ),

          retries:
            nonNegativeInteger(
              value.telemetry
                .retries,
              "calibration.telemetry.retries",
            ),

          fallbackCount:
            nonNegativeInteger(
              value.telemetry
                .fallbackCount,
              "calibration.telemetry.fallbackCount",
            ),
        }
      : undefined,

    provenance:
      value.provenance
        ? structuredClone(
            value.provenance,
          )
        : undefined,
  };
}

function estimateKey(
  input: {
    modelId: string;
    modelVersion: string;
    strategy?: string;
    domain: string;
  },
): string {
  return [
    input.modelId,
    input.modelVersion,
    input.strategy ??
      "any-strategy",
    input.domain,
  ].join(
    "::",
  );
}

function recencyWeight(
  observedAt: string,
  now: Date,
  halfLifeDays: number,
): number {
  const ageMs =
    Math.max(
      0,
      now.getTime() -
      new Date(
        observedAt,
      ).getTime(),
    );

  const ageDays =
    ageMs /
    DAY_MS;

  return Math.pow(
    0.5,
    ageDays /
    halfLifeDays,
  );
}

function sourceWeight(
  source:
    CalibrationObservationSource,
  policy:
    ContinuousCalibrationPolicy,
): number {
  return source ===
    "scheduled-evaluation"
    ? policy.sourceWeights
        .scheduledEvaluation
    : policy.sourceWeights
        .realWork;
}

function observationWeight(
  observation:
    CalibrationObservation,
  now: Date,
  policy:
    ContinuousCalibrationPolicy,
): number {
  const confidence =
    observation.objective
      .evidenceConfidence ??
    0.75;

  return (
    recencyWeight(
      observation.observedAt,
      now,
      policy.halfLifeDays,
    ) *
    sourceWeight(
      observation.source,
      policy,
    ) *
    confidence
  );
}

function weightedAverage(
  values:
    Array<{
      value: number;
      weight: number;
    }>,
): number | undefined {
  const usable =
    values.filter(
      (entry) =>
        Number.isFinite(
          entry.value,
        ) &&
        Number.isFinite(
          entry.weight,
        ) &&
        entry.weight >
        0,
    );

  if (
    usable.length ===
    0
  ) {
    return undefined;
  }

  const totalWeight =
    usable.reduce(
      (
        total,
        entry,
      ) =>
        total +
        entry.weight,
      0,
    );

  if (
    totalWeight <=
    0
  ) {
    return undefined;
  }

  return (
    usable.reduce(
      (
        total,
        entry,
      ) =>
        total +
        entry.value *
          entry.weight,
      0,
    ) /
    totalWeight
  );
}

function binaryRate(
  values:
    Array<{
      value: boolean;
      weight: number;
    }>,
): number | undefined {
  return weightedAverage(
    values.map(
      (entry) => ({
        value:
          entry.value
            ? 1
            : 0,

        weight:
          entry.weight,
      }),
    ),
  );
}

function estimateConfidence(
  effectiveSamples: number,
  policy:
    ContinuousCalibrationPolicy,
): number {
  const ratio =
    Math.min(
      1,
      effectiveSamples /
      policy.strongSampleCount,
    );

  return Math.max(
    0,
    Math.min(
      1,
      0.25 +
      0.75 *
      ratio,
    ),
  );
}

function dueReasons(
  input: {
    lastScheduledEvaluationAt?:
      string;

    effectiveSampleCount:
      number;

    now: Date;

    policy:
      ContinuousCalibrationPolicy;
  },
): string[] {
  const reasons:
    string[] =
      [];

  if (
    input.effectiveSampleCount <
    input.policy
      .minimumEffectiveSamples
  ) {
    reasons.push(
      "insufficient-effective-samples",
    );
  }

  if (
    !input
      .lastScheduledEvaluationAt
  ) {
    reasons.push(
      "no-scheduled-evaluation-for-current-version",
    );

    return reasons;
  }

  const ageDays =
    Math.max(
      0,
      input.now.getTime() -
      new Date(
        input
          .lastScheduledEvaluationAt,
      ).getTime(),
    ) /
    DAY_MS;

  if (
    ageDays >
    input.policy
      .scheduledEvaluationIntervalDays
  ) {
    reasons.push(
      "scheduled-evaluation-stale",
    );
  }

  return reasons;
}

export class ContinuousCalibrationEngine {
  private readonly observations =
    new Map<
      string,
      CalibrationObservation
    >();

  private readonly policy:
    ContinuousCalibrationPolicy;

  private readonly filePath:
    string;

  private readonly clock:
    () => Date;

  constructor(
    options: {
      policy?:
        Partial<
          ContinuousCalibrationPolicy
        >;

      filePath?:
        string;

      clock?:
        () => Date;
    } = {},
  ) {
    this.policy =
      validatePolicy({
        ...DEFAULT_CONTINUOUS_CALIBRATION_POLICY,

        ...(
          options.policy ??
          {}
        ),

        sourceWeights: {
          ...DEFAULT_CONTINUOUS_CALIBRATION_POLICY
            .sourceWeights,

          ...(
            options.policy
              ?.sourceWeights ??
            {}
          ),
        },
      });

    this.filePath =
      options.filePath ??
      join(
        process.cwd(),
        ".chernobog",
        "cognition",
        "continuous-calibration.json",
      );

    this.clock =
      options.clock ??
      (() => new Date());
  }

  get size(): number {
    return this.observations
      .size;
  }

  list():
    CalibrationObservation[] {
    return [
      ...this.observations
        .values(),
    ]
      .sort(
        (
          left,
          right,
        ) =>
          left.observedAt
            .localeCompare(
              right.observedAt,
            ) ||
          left.id
            .localeCompare(
              right.id,
            ),
      )
      .map(
        (entry) =>
          structuredClone(
            entry,
          ),
      );
  }

  clear(): void {
    this.observations
      .clear();
  }

  ingest(
    observation:
      CalibrationObservation,
  ):
    CalibrationObservation {
    const normalized =
      normalizeObservation(
        observation,
      );

    this.observations
      .set(
        normalized.id,
        normalized,
      );

    this.trim();

    return structuredClone(
      normalized,
    );
  }

  ingestMany(
    observations:
      readonly CalibrationObservation[],
  ):
    CalibrationObservation[] {
    return observations
      .map(
        (observation) =>
          this.ingest(
            observation,
          ),
      );
  }

  estimate(
    subject:
      CalibrationSubject,
    domain: string,
    now =
      this.clock(),
  ):
    CapabilityEstimate {
    const modelId =
      normalizeText(
        subject.modelId,
        "calibrationSubject.modelId",
      );

    const modelVersion =
      normalizeText(
        subject.modelVersion,
        "calibrationSubject.modelVersion",
      );

    const strategy =
      optionalText(
        subject.strategy,
      );

    const normalizedDomain =
      normalizeDomain(
        domain,
      );

    const matching =
      this.list()
        .filter(
          (observation) =>
            observation.modelId ===
              modelId &&
            observation.modelVersion ===
              modelVersion &&
            (
              strategy ===
                undefined ||
              observation.strategy ===
                strategy
            ) &&
            observation.task.domain ===
              normalizedDomain,
        );

    const weighted =
      matching.map(
        (observation) => ({
          observation,

          weight:
            observationWeight(
              observation,
              now,
              this.policy,
            ),
        }),
      );

    const totalWeight =
      weighted.reduce(
        (
          total,
          entry,
        ) =>
          total +
          entry.weight,
        0,
      );

    const objectiveScore =
      weightedAverage(
        weighted.map(
          (entry) => ({
            value:
              entry.observation
                .objective
                .score,

            weight:
              entry.weight,
          }),
        ),
      ) ??
      0.5;

    const scheduled =
      matching.filter(
        (observation) =>
          observation.source ===
          "scheduled-evaluation",
      );

    const lastScheduledEvaluationAt =
      scheduled.length >
        0
        ? scheduled[
            scheduled.length -
            1
          ]?.observedAt
        : undefined;

    const reasons =
      dueReasons({
        lastScheduledEvaluationAt,

        effectiveSampleCount:
          totalWeight,

        now,

        policy:
          this.policy,
      });

    const subjectiveEntries =
      weighted.filter(
        (entry) =>
          entry.observation
            .subjective
            ?.preferenceScore !==
            undefined ||
          entry.observation
            .subjective
            ?.userRating !==
            undefined,
      );

    return {
      key:
        estimateKey({
          modelId,
          modelVersion,
          strategy,
          domain:
            normalizedDomain,
        }),

      modelId,
      modelVersion,
      strategy,
      domain:
        normalizedDomain,

      objectiveScore:
        Math.max(
          0,
          Math.min(
            1,
            objectiveScore,
          ),
        ),

      confidence:
        estimateConfidence(
          totalWeight,
          this.policy,
        ),

      sampleCount:
        matching.length,

      effectiveSampleCount:
        totalWeight,

      sourceCounts: {
        realWork:
          matching.filter(
            (observation) =>
              observation.source ===
              "real-work",
          ).length,

        scheduledEvaluation:
          scheduled.length,
      },

      verification: {
        passRate:
          binaryRate(
            weighted
              .filter(
                (entry) =>
                  entry.observation
                    .objective
                    .verificationVerdict !==
                    undefined &&
                  entry.observation
                    .objective
                    .verificationVerdict !==
                    "unknown",
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .objective
                      .verificationVerdict ===
                    "pass",

                  weight:
                    entry.weight,
                }),
              ),
          ),

        rejectRate:
          binaryRate(
            weighted
              .filter(
                (entry) =>
                  entry.observation
                    .objective
                    .verificationVerdict !==
                    undefined &&
                  entry.observation
                    .objective
                    .verificationVerdict !==
                    "unknown",
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .objective
                      .verificationVerdict ===
                    "reject",

                  weight:
                    entry.weight,
                }),
              ),
          ),
      },

      executionSuccessRate:
        binaryRate(
          weighted
            .filter(
              (entry) =>
                entry.observation
                  .objective
                  .executionSucceeded !==
                undefined,
            )
            .map(
              (entry) => ({
                value:
                  entry.observation
                    .objective
                    .executionSucceeded ===
                  true,

                weight:
                  entry.weight,
              }),
            ),
        ),

      telemetry: {
        weightedLatencyMs:
          weightedAverage(
            weighted
              .filter(
                (entry) =>
                  entry.observation
                    .telemetry
                    ?.latencyMs !==
                  undefined,
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .telemetry
                      ?.latencyMs ??
                    0,

                  weight:
                    entry.weight,
                }),
              ),
          ),

        weightedCost:
          weightedAverage(
            weighted
              .filter(
                (entry) =>
                  entry.observation
                    .telemetry
                    ?.cost !==
                  undefined,
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .telemetry
                      ?.cost ??
                    0,

                  weight:
                    entry.weight,
                }),
              ),
          ),

        weightedRetries:
          weightedAverage(
            weighted
              .filter(
                (entry) =>
                  entry.observation
                    .telemetry
                    ?.retries !==
                  undefined,
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .telemetry
                      ?.retries ??
                    0,

                  weight:
                    entry.weight,
                }),
              ),
          ),

        weightedFallbackCount:
          weightedAverage(
            weighted
              .filter(
                (entry) =>
                  entry.observation
                    .telemetry
                    ?.fallbackCount !==
                  undefined,
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .telemetry
                      ?.fallbackCount ??
                    0,

                  weight:
                    entry.weight,
                }),
              ),
          ),
      },

      subjective: {
        sampleCount:
          subjectiveEntries
            .length,

        weightedPreferenceScore:
          weightedAverage(
            subjectiveEntries
              .filter(
                (entry) =>
                  entry.observation
                    .subjective
                    ?.preferenceScore !==
                  undefined,
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .subjective
                      ?.preferenceScore ??
                    0,

                  weight:
                    entry.weight,
                }),
              ),
          ),

        weightedUserRating:
          weightedAverage(
            subjectiveEntries
              .filter(
                (entry) =>
                  entry.observation
                    .subjective
                    ?.userRating !==
                  undefined,
              )
              .map(
                (entry) => ({
                  value:
                    entry.observation
                      .subjective
                      ?.userRating ??
                    0,

                  weight:
                    entry.weight,
                }),
              ),
          ),
      },

      firstObservedAt:
        matching[0]
          ?.observedAt ??
        now.toISOString(),

      lastObservedAt:
        matching[
          matching.length -
          1
        ]?.observedAt ??
        now.toISOString(),

      lastScheduledEvaluationAt,

      recalibration: {
        due:
          reasons.length >
          0,

        reasons,
      },
    };
  }

  snapshot(
    subjects:
      readonly CalibrationSubject[],
    now =
      this.clock(),
  ):
    CalibrationSnapshot {
    const estimates:
      CapabilityEstimate[] =
        [];

    for (
      const subject
      of subjects
    ) {
      const domains =
        [
          ...new Set(
            (
              subject.domains ??
              []
            )
              .map(
                normalizeDomain,
              ),
          ),
        ];

      const observedDomains =
        this.list()
          .filter(
            (observation) =>
              observation.modelId ===
                subject.modelId &&
              observation.modelVersion ===
                subject.modelVersion &&
              (
                subject.strategy ===
                  undefined ||
                observation.strategy ===
                  subject.strategy
              ),
          )
          .map(
            (observation) =>
              observation.task
                .domain,
          );

      const allDomains =
        [
          ...new Set(
            [
              ...domains,
              ...observedDomains,
            ],
          ),
        ]
          .sort();

      for (
        const domain
        of allDomains
      ) {
        estimates.push(
          this.estimate(
            subject,
            domain,
            now,
          ),
        );
      }
    }

    estimates.sort(
      (
        left,
        right,
      ) =>
        left.key
          .localeCompare(
            right.key,
          ),
    );

    return {
      version:
        CONTINUOUS_CALIBRATION_VERSION,

      generatedAt:
        now.toISOString(),

      policy:
        structuredClone(
          this.policy,
        ),

      observationCount:
        this.observations
          .size,

      estimates:
        structuredClone(
          estimates,
        ),

      dueSubjects:
        estimates
          .filter(
            (estimate) =>
              estimate
                .recalibration
                .due,
          )
          .map(
            (estimate) => ({
              modelId:
                estimate.modelId,

              modelVersion:
                estimate
                  .modelVersion,

              strategy:
                estimate.strategy,

              domain:
                estimate.domain,

              reasons:
                structuredClone(
                  estimate
                    .recalibration
                    .reasons,
                ),
            }),
          ),

      invariants: {
        subjectivePreferenceAffectsObjectiveScore:
          false,

        crossVersionEvidenceMixed:
          false,

        routingMutated:
          false,

        registryMutated:
          false,

        scheduledEvaluationExecuted:
          false,
      },
    };
  }

  async save(
    now =
      this.clock(),
  ):
    Promise<void> {
    const snapshot:
      CalibrationSnapshotFile = {
        schemaVersion:
          1,

  }

          observation.id,
        );

export type ModelRole =

    | "env"

export type ModelCandidateSource =
  | "role-primary"
  | "builtin-role"
  | "default-model"
  | "builtin-default";

export interface ModelCandidate {
  role: ModelRole;
  model: string;
  source:
    ModelCandidateSource;
}

const BUILTIN_DEFAULT_MODEL =
  "gemma3";

const BUILTIN_CODE_MODEL =
  "deepseek-coder-v2:16b";

function readEnv(
  name: string,
) {
  const value =
    process.env[name];

  if (
    typeof value !== "string"
  ) {
    return undefined;
  }

  const trimmed =
    value.trim();

  return trimmed.length > 0
    ? trimmed
    : undefined;
}

export function resolveModel(
  role:
    ModelRole = "default",
): ResolvedModel {
  const defaultModel =
    readEnv("OLLAMA_MODEL") ??
    BUILTIN_DEFAULT_MODEL;

  const codeModel =
    readEnv(
      "OLLAMA_CODE_MODEL",
    ) ??
    readEnv("OLLAMA_MODEL") ??
    BUILTIN_CODE_MODEL;

  switch (role) {
    case "code":
      return {
        role,
        model:
          codeModel,
        source:
          readEnv(
            "OLLAMA_CODE_MODEL",
          )
            ? "env"
            : "fallback",
      };

    case "planner":
      return {
        role,
        model:
          codeModel,
        source:
          readEnv(
            "OLLAMA_CODE_MODEL",
          )
            ? "env"
            : "fallback",
      };

    case "repair":
      return {
        role,
        model:
          codeModel,
        source:
          readEnv(
            "OLLAMA_CODE_MODEL",
          )
            ? "env"
            : "fallback",
      };

    case "default":
    default:
      return {
        role:
          "default",
        model:
          defaultModel,
        source:
          readEnv(
            "OLLAMA_MODEL",
          )
            ? "env"
            : "fallback",
      };
  }
}

function dedupeCandidates(
  candidates:
    readonly ModelCandidate[],
): ModelCandidate[] {
  const seen =
    new Set<string>();

  const output:
    ModelCandidate[] = [];

  for (
    const candidate
    of candidates
  ) {
    const normalized =
      normalizeOllamaModelName(
        candidate.model,
      );

    if (
      !normalized ||
      seen.has(normalized)
    ) {
      continue;
    }

    seen.add(normalized);

    output.push({
      ...candidate,
    });
  }

  return output;
}

export function getModelCandidates(
  role:
    ModelRole = "default",
): ModelCandidate[] {
  const primary =
    resolveModel(role);

  const defaultModel =
    resolveModel("default");

  if (role === "default") {
    return dedupeCandidates([
      {
        role,
        model:
          primary.model,
        source:
          "role-primary",
      },
      {
        role,
        model:
          BUILTIN_DEFAULT_MODEL,
        source:
          "builtin-default",
      },
    ]);
  }

  return dedupeCandidates([
    {
      role,
      model:
        primary.model,
      source:
        "role-primary",
    },
    {
      role,
      model:
        BUILTIN_CODE_MODEL,
      source:
        "builtin-role",
    },
    {
      role,
      model:
        defaultModel.model,
      source:
        "default-model",
    },
    {
      role,
      model:
        BUILTIN_DEFAULT_MODEL,
      source:
        "builtin-default",
    },
  ]);
}

    boolean;

    boolean;

    boolean;

    | "complexity-fit"
    | "reasoning-fit"
    | "factuality-fit"
    | "creativity-fit"
    | "tool-fit"
    | "cost-fit"
    | "latency-fit"

}

}

    RankedStrategy[];

    ];

      string;

  };

    StrategyRoutingConstraints;

    string[];
}

function clamp01(
  value: number,
): number {
  if (
    !Number.isFinite(
      value,
    )
  ) {
    return 0;
  }

  return Math.max(
    0,
    Math.min(
      1,
      value,
    ),
  );
}

function setOf<T extends string>(
  values:
    readonly T[]
    | undefined,
): Set<T> {
  return new Set(
    values ?? [],
  );
}

function selectedModelConfidence(
  modelRoute:
    ModelRouteDecision,
): number {
  return clamp01(
    modelRoute
      .selected
      ?.confidence ??
    0,
  );
}

function eligibleModels(
  modelRoute:
    ModelRouteDecision,
): RankedModelCandidate[] {
  return modelRoute
    .ranking
    .filter(
      (entry) =>
        entry.eligible,
    );
}

function scoreComponent(
  component:
    StrategyScoreComponent[
      "component"
    ],
  score:
    number,
  weight:
    number,
  reasons:
    string[],
): StrategyScoreComponent {
  const normalized =
    clamp01(
      score,
    );

  return {
    component,
    score:
      normalized,
    weight,
    weightedScore:
      normalized *
      weight,
    reasons,
  };
}

          candidateCount >
          1

          "Specialist routing is useful for domain-specific work when multiple eligible models exist.",

          complexity *

          reasoning *
            0.3,

          factuality *
            0.25,

            0.25,

function recipeFor(
  strategy:
    CognitiveStrategyKind,
  interpretation:
    InputInterpretation,
  constraints:
    StrategyRoutingConstraints,
): StrategyRecipe {
  const capped =
    Math.max(
      1,
      Math.min(
        8,
        Math.trunc(
          constraints.maxPasses ??
          4,
        ),
      ),
    );

  switch (
    strategy
  ) {
    case "single-model":
      return {
        strategy,
        multiPass:
          false,
        multiModel:
          false,
        requiresTools:
          false,
        maxPasses:
          1,
        steps: [
          {
            order:
              1,
            action:
              "reason",
            purpose:
              "Produce the answer with the selected primary model.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
        ],
      };

    case "cheap-first-escalation":
      return {
        strategy,
        multiPass:
          true,
        multiModel:
          true,
        requiresTools:
          false,
        maxPasses:
          Math.min(
            capped,
            2,
          ),
        steps: [
          {
            order:
              1,
            action:
              "reason",
            purpose:
              "Attempt the task with a lower-cost eligible model.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
          {
            order:
              2,
            action:
              "escalate",
            purpose:
              "Escalate only if confidence/validation is insufficient.",
            modelRole:
              "fallback",
            toolUseAllowed:
              false,
          },
        ],
      };

    case "specialist":
      return {
        strategy,
        multiPass:
          false,
        multiModel:
          false,
        requiresTools:
          false,
        maxPasses:
          1,
        steps: [
          {
            order:
              1,
            action:
              "reason",
            purpose:
              `Use a domain-specialist model for ${interpretation.taskIntelligence.primaryDomain.value}.`,
            modelRole:
              "specialist",
            toolUseAllowed:
              false,
          },
        ],
      };

    case "competition":
      return {
        strategy,
        multiPass:
          true,
        multiModel:
          true,
        requiresTools:
          false,
        maxPasses:
          Math.min(
            capped,
            3,
          ),
        steps: [
          {
            order:
              1,
            action:
              "reason",
            purpose:
              "Generate an independent candidate with the primary model.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
          {
            order:
              2,
            action:
              "reason",
            purpose:
              "Generate an independent competing candidate.",
            modelRole:
              "secondary",
            toolUseAllowed:
              false,
          },
          {
            order:
              3,
            action:
              "compare",
            purpose:
              "Compare candidates using objective task criteria.",
            modelRole:
              "critic",
            toolUseAllowed:
              false,
          },
        ],
      };

    case "critic-loop":
      return {
        strategy,
        multiPass:
          true,
        multiModel:
          false,
        requiresTools:
          false,
        maxPasses:
          Math.min(
            capped,
            3,
          ),
        steps: [
          {
            order:
              1,
            action:
              "reason",
            purpose:
              "Produce an initial candidate.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
          {
            order:
              2,
            action:
              "critique",
            purpose:
              "Critique the candidate against factual, logical, and constraint requirements.",
            modelRole:
              "critic",
            toolUseAllowed:
              false,
          },
          {
            order:
              3,
            action:
              "refine",
            purpose:
              "Repair issues identified by the critique.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
        ],
      };

    case "refinement":
      return {
        strategy,
        multiPass:
          true,
        multiModel:
          false,
        requiresTools:
          false,
        maxPasses:
          Math.min(
            capped,
            3,
          ),
        steps: [
          {
            order:
              1,
            action:
              "reason",
            purpose:
              "Produce the first complete candidate.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
          {
            order:
              2,
            action:
              "refine",
            purpose:
              "Improve quality, clarity, completeness, and task fit.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
        ],
      };

    case "tool-assisted":
      return {
        strategy,
        multiPass:
          true,
        multiModel:
          false,
        requiresTools:
          true,
        maxPasses:
          Math.min(
            capped,
            3,
          ),
        steps: [
          {
            order:
              1,
            action:
              "use-tools",
            purpose:
              "Acquire required external evidence or tool results before concluding.",
            modelRole:
              "primary",
            toolUseAllowed:
              true,
          },
          {
            order:
              2,
            action:
              "reason",
            purpose:
              "Reason over authoritative tool results and task context.",
            modelRole:
              "primary",
            toolUseAllowed:
              true,
          },
          {
            order:
              3,
            action:
              "refine",
            purpose:
              "Produce the final grounded answer.",
            modelRole:
              "primary",
            toolUseAllowed:
              false,
          },
        ],
      };
  }
}

function eligibilityReasons(
  strategy:
    CognitiveStrategyKind,
  interpretation:
    InputInterpretation,
  modelRoute:
    ModelRouteDecision,
  constraints:
    StrategyRoutingConstraints,
): string[] {
  const reasons:
    string[] =
      [];

  const allowed =
    setOf(
      constraints
        .allowedStrategies,
    );

  const excluded =
    setOf(
      constraints
        .excludedStrategies,
    );

  if (
    allowed.size >
      0 &&
    !allowed.has(
      strategy,
    )
  ) {
    reasons.push(
      "strategy is not in allowedStrategies",
    );
  }

  if (
    excluded.has(
      strategy,
    )
  ) {
    reasons.push(
      "strategy is explicitly excluded",
    );
  }

  const recipe =
    recipeFor(
      strategy,
      interpretation,
      constraints,
    );

  if (
    constraints
      .forbidMultiModel &&
    recipe.multiModel
  ) {
    reasons.push(
      "multi-model strategies are forbidden",
    );
  }

  if (
    constraints
      .requireToolAssistance &&
    !recipe.requiresTools
  ) {
    reasons.push(
      "tool assistance is explicitly required",
    );
  }

  if (
    recipe.multiModel &&
    eligibleModels(
      modelRoute,
    ).length <
      2
  ) {
    reasons.push(
      "strategy requires at least two eligible models",
    );
  }

  if (
    strategy ===
      "tool-assisted" &&
    !interpretation
      .taskIntelligence
      .derived
      .toolUseRequired &&
    !constraints
      .requireToolAssistance
  ) {
    reasons.push(
      "task does not require tool assistance",
    );
  }

  if (
    modelRoute.status !==
      "selected"
  ) {
    reasons.push(
      "no primary model has been selected by OH-4M",
    );
  }

  return reasons;
}

function rankStrategy(
  strategy:
    CognitiveStrategyKind,
  interpretation:
    InputInterpretation,
  modelRoute:
    ModelRouteDecision,
  constraints:
    StrategyRoutingConstraints,
): RankedStrategy {
  const base =
    strategyBaseScores(
      strategy,
      interpretation,
      modelRoute,
    );

  const weights = {
    complexity:
      0.15,
    reasoning:
      0.16,
    factuality:
      0.16,
    creativity:
      0.1,
    tools:
      0.13,
    cost:
      0.1,
    latency:
      0.1,
    modelConfidence:
      0.1,
  };

  const components: StrategyScoreComponent[] =
    [
      scoreComponent(
        "complexity-fit",
        base.complexity,
        weights.complexity,
        [
          `Task complexity=${interpretation.taskIntelligence.complexity.value}.`,
        ],
      ),
      scoreComponent(
        "reasoning-fit",
        base.reasoning,
        weights.reasoning,
        [
          `Reasoning depth=${interpretation.taskIntelligence.reasoningDepth.value}.`,
        ],
      ),
      scoreComponent(
        "factuality-fit",
        base.factuality,
        weights.factuality,
        [
          `Factuality=${interpretation.taskIntelligence.factuality.value}.`,
        ],
      ),
      scoreComponent(
        "creativity-fit",
        base.creativity,
        weights.creativity,
        [
          `Creativity=${interpretation.taskIntelligence.creativity.value}.`,
        ],
      ),
      scoreComponent(
        "tool-fit",
        base.tools,
        weights.tools,
        [
          `Tool use required=${String(interpretedToolUse(interpretedToolUseInput(interpretation)))}.`,
        ],
      ),
      scoreComponent(
        "cost-fit",
        base.cost,
        weights.cost,
        [
          `Cost tolerance=${interpretation.taskIntelligence.costTolerance.value}.`,
        ],
      ),
      scoreComponent(
        "latency-fit",
        base.latency,
        weights.latency,
        [
          `Latency tolerance=${interpretation.taskIntelligence.latencyTolerance.value}.`,
        ],
      ),
      scoreComponent(
        "model-confidence-fit",
        base.modelConfidence,
        weights.modelConfidence,
        [
          `OH-4M selected-model confidence=${selectedModelConfidence(modelRoute).toFixed(2)}.`,
        ],
      ),
    ];

  const exclusions =
    eligibilityReasons(
      strategy,
      interpretation,
      modelRoute,
      constraints,
    );

  const rawScore =
    components.reduce(
      (
        total,
        entry,
      ) =>
        total +
        entry.weightedScore,
      0,
    );

  const confidence =
    clamp01(
      0.55 +
      selectedModelConfidence(
        modelRoute,
      ) *
        0.3 +
      (
        interpretation
          .taskIntelligence
          .primaryDomain
          .confidence ??
        0
      ) *
        0.15,
    );

  return {
    strategy,
    eligible:
      exclusions.length ===
      0,
    exclusionReasons:
      exclusions,
    score:
      exclusions.length >
        0
        ? 0
        : clamp01(
            rawScore,
          ),
    confidence,
    components,
    recipe:
      recipeFor(
        strategy,
        interpretation,
        constraints,
      ),
    reasons: [
      ...base.reasons,
      ...components.flatMap(
        (entry) =>
          entry.reasons.map(
            (reason) =>
              `${entry.component}: ${reason}`,
          ),
      ),
    ],
  };
}

function interpretedToolUseInput(
  interpretation:
    InputInterpretation,
): InputInterpretation {
  return interpretation;
}

function interpretedToolUse(
  interpretation:
    InputInterpretation,
): boolean {
  return interpretation
    .taskIntelligence
    .derived
    .toolUseRequired;
}

const ALL_STRATEGIES:
  CognitiveStrategyKind[] =
    [
      "single-model",
      "cheap-first-escalation",
      "specialist",
      "competition",
      "critic-loop",
      "refinement",
      "tool-assisted",
    ];

export class StrategyRouter {
  route(
    input: {
      interpretation:
        InputInterpretation;
      modelRoute:
        ModelRouteDecision;
      constraints?:
        StrategyRoutingConstraints;
    },
  ): StrategyRouteDecision {
    const constraints =
      structuredClone(
        input.constraints ??
        {},
      );

    const ranking =
      ALL_STRATEGIES
        .map(
          (strategy) =>
            rankStrategy(
              strategy,
              input.interpretation,
              input.modelRoute,
              constraints,
            ),
        )
        .sort(
          (
            left,
            right,
          ) => {
            if (
              left.eligible !==
              right.eligible
            ) {
              return left.eligible
                ? -1
                : 1;
            }

            if (
              left.score !==
              right.score
            ) {
              return (
                right.score -
                left.score
              );
            }

            if (
              left.confidence !==
              right.confidence
            ) {
              return (
                right.confidence -
                left.confidence
              );
            }

            return left.strategy
              .localeCompare(
                right.strategy,
              );
          },
        );

    const minimum =
      clamp01(
        constraints
          .minimumStrategyConfidence ??
        0,
      );

    const selected =
      ranking.find(
        (entry) =>
          entry.eligible &&
          entry.confidence >=
            minimum,
      );

    const warnings:
      string[] =
        [];

    if (
      input.modelRoute
        .status !==
      "selected"
    ) {
      warnings.push(
        "OH-4M did not select a primary model; no strategy may execute safely.",
      );
    }

    if (!selected) {
      warnings.push(
        "No eligible cognitive strategy satisfied the routing constraints/confidence threshold.",
      );
    }

    if (
      selected
        ?.recipe.multiModel
    ) {
      warnings.push(
        "Selected strategy requires multiple models; execution remains deferred to OH-4S Multi-Model Orchestrator.",
      );
    }

    if (
      selected
        ?.recipe.requiresTools
    ) {
      warnings.push(
        "Selected strategy requires tools; execution remains subject to later authority/action bridges.",
      );
    }

    return {
      version:
        STRATEGY_ROUTER_VERSION,
      routedAt:
        new Date()
          .toISOString(),
      status:
        selected
          ? "selected"
          : "no-eligible-strategy",
      selected:
        selected
          ? structuredClone(
              selected,
            )
          : undefined,
      ranking,
      modelRouteStatus:
        input.modelRoute
          .status,
      selectedModelId:
        input.modelRoute
          .selected
          ?.candidate.id,
      taskSummary: {
        primaryDomain:
          input
            .interpretation
            .taskIntelligence
            .primaryDomain
            .value,
        complexity:
          input
            .interpretation
            .taskIntelligence
            .complexity
            .value,
        creativity:
          input
            .interpretation
            .taskIntelligence
            .creativity
            .value,
        factuality:
          input
            .interpretation
            .taskIntelligence
            .factuality
            .value,
        reasoningDepth:
          input
            .interpretation
            .taskIntelligence
            .reasoningDepth
            .value,
        latencyTolerance:
          input
            .interpretation
            .taskIntelligence
            .latencyTolerance
            .value,
        costTolerance:
          input
            .interpretation
            .taskIntelligence
            .costTolerance
            .value,
        toolUseRequired:
          input
            .interpretation
            .taskIntelligence
            .derived
            .toolUseRequired,
      },
      constraints,
      warnings,
    };
  }
}

import type {
  ModelGateway,
  ModelRequest,
  ModelResult,
} from "@chernobog/contracts/models";

import {
  generateWithReliableOllama,
} from "./reliableOllama";
import {
  recordModelInvocationTelemetry,
  unavailableCostUsage,
  unavailableTokenUsage,
  type ModelInvocationTelemetryContext,
} from "./telemetry";

export type ModelGenerateOptions =
  Parameters<
    typeof generateWithReliableOllama
  >[0] & {
    telemetry?:
      ModelInvocationTelemetryContext;
  };

export type ModelGenerateResult =
  Awaited<
    ReturnType<
      typeof generateWithReliableOllama
    >
  >;

/**
 * Provider-neutral public generation entry point.
 * This preserves the accepted low-level generation semantics while hiding
 * the current provider from business/project callers.
 */
export async function generateWithModel(
  options: ModelGenerateOptions,
): Promise<ModelGenerateResult> {
  const {
    telemetry,
    ...generationOptions
  } = options;

  const startedAt =
    Date.now();

  try {
    const result =
      await generateWithReliableOllama(
        generationOptions,
      );

    const attempts =
      result.attempts ??
      1;

    await recordModelInvocationTelemetry(
      {
        taskType:
          telemetry?.taskType ??
          result.role ??
          "unspecified",
        provider:
          "ollama",
        model:
          result.model,
        requestedRole:
          generationOptions.role ??
          "default",
        selectedModel:
          result.selection
            ?.selectedInstalledModel ??
          result.selection
            ?.selectedModel ??
          result.model,
        selectionReason:
          result.selection
            ?.reason,
        outcome:
          result.ok
            ? "success"
            : "failure",
        latencyMs:
          result.durationMs ??
          Date.now() -
            startedAt,
        attempts,
        retries:
          Math.max(
            0,
            attempts - 1,
          ),
        fallbackCount:
          result.fallbackCount ??
          0,
        tokenUsage:
          unavailableTokenUsage(),
        cost:
          unavailableCostUsage(),
        validationExpected:
          telemetry
            ?.validationExpected ===
          true,
        error:
          result.error,
        failureKind:
          result.failureKind,
      },
      telemetry,
    );

    return result;
  } catch (error) {
    await recordModelInvocationTelemetry(
      {
        taskType:
          telemetry?.taskType ??
          generationOptions.role ??
          "unspecified",
        provider:
          "ollama",
        model:
          generationOptions
            .modelOverride ??
          "unresolved",
        requestedRole:
          generationOptions.role ??
          "default",
        outcome:
          "failure",
        latencyMs:
          Date.now() -
          startedAt,
        attempts:
          0,
        retries:
          0,
        fallbackCount:
          0,
        tokenUsage:
          unavailableTokenUsage(),
        cost:
          unavailableCostUsage(),
        validationExpected:
          telemetry
            ?.validationExpected ===
          true,
        error:
          error instanceof Error
            ? error.message
            : String(error),
        failureKind:
          "gateway-exception",
      },
      telemetry,
    );

    throw error;
  }
}

export const modelGateway: ModelGateway = {
  async execute<TOutput = unknown>(
    request: ModelRequest,
  ): Promise<ModelResult<TOutput>> {
    const metadata =
      request.metadata ?? {};

    const result =
      await generateWithModel({
        role:
          (request.role ?? "default") as any,
        ...(request.prompt
          ? { prompt: request.prompt }
          : {}),
        ...(request.messages
          ? { messages: request.messages as any }
          : {}),
        ...(typeof metadata.temperature === "number"
          ? { temperature: metadata.temperature }
          : {}),
        ...(typeof metadata.timeoutMs === "number"
          ? { timeoutMs: metadata.timeoutMs }
          : {}),
        ...(typeof metadata.numPredict === "number"
          ? { numPredict: metadata.numPredict }
          : {}),
      });

    return {
      ok: result.ok,
      ...(result.text !== undefined
        ? { output: result.text as TOutput }
        : {}),
      ...(result.error
        ? { error: result.error }
        : {}),
      provider: "ollama",
      model: result.model,
      metadata: {
        role: result.role,
      },
    };
  },
};
