import {
  NextResponse,
} from "next/server";

import {
  getAutonomousCognitiveLoopRuntime,
  type AutonomousCognitiveLoopInput,
} from "@chernobog/models/autonomousCognitiveLoop";

export const runtime =
  "nodejs";

function failure(
  error: unknown,
  status = 400,
) {
  return NextResponse.json(
    {
      ok:
        false,

      error:
        error instanceof Error
          ? error.message
          : String(error),
    },
    {
      status,
    },
  );
}

export async function POST(
  request: Request,
) {
  try {
    const body =
      await request.json() as {
        action?:
          "start"
          | "run"
          | "resume"
          | "interrupt";

        id?: string;

        input?:
          AutonomousCognitiveLoopInput;

        humanInput?: unknown;

        reason?: string;
      };

    const loop =
      getAutonomousCognitiveLoopRuntime();

    switch (
      body.action
    ) {
      case "start":
        if (
          !body.input
        ) {
          throw new Error(
            "input is required for OH-4AD start.",
          );
        }

        return NextResponse.json({
          ok:
            true,

          state:
            await loop.start(
              body.input,
            ),
        });

      case "run":
        if (
          !body.id
        ) {
          throw new Error(
            "id is required for OH-4AD run.",
          );
        }

        return NextResponse.json({
          ok:
            true,

          state:
            await loop.run(
              body.id,
            ),
        });

      case "resume":
        if (
          !body.id
        ) {
          throw new Error(
            "id is required for OH-4AD resume.",
          );
        }

        return NextResponse.json({
          ok:
            true,

          state:
            await loop.resume(
              body.id,
              {
                humanInput:
                  body.humanInput,
              },
            ),
        });

      case "interrupt":
        if (
          !body.id
        ) {
          throw new Error(
            "id is required for OH-4AD interrupt.",
          );
        }

        return NextResponse.json({
          ok:
            true,

          state:
            await loop.interrupt(
              body.id,
              body.reason ??
              "Interrupted by caller.",
            ),
        });

      default:
        throw new Error(
          "action must be start, run, resume, or interrupt.",
        );
    }
  } catch (error) {
    return failure(
      error,
    );
  }
}

export async function GET(
  request: Request,
) {
  try {
    const url =
      new URL(
        request.url,
      );

    const id =
      url.searchParams
        .get(
          "id",
        )
        ?.trim();

    if (!id) {
      throw new Error(
        "id query parameter is required.",
      );
    }

    const state =
      await getAutonomousCognitiveLoopRuntime()
        .load(
          id,
        );

    return NextResponse.json({
      ok:
        true,

      state,
    });
  } catch (error) {
    return failure(
      error,
    );
  }
}

function recoveryResumeToStage(
  resumeStage:
    RecoveryResumeStage,
  action:
    RecoveryActionKind,
  failedStage:
    AutonomousCognitiveStage,
): AutonomousCognitiveStage | undefined {
  if (
    action ===
      "retry-tool" ||
    action ===
      "resume-from-checkpoint"
  ) {
    return "act";
  }

  switch (
    resumeStage
  ) {
    case "context":
      return "contextualize";

    case "prompt":
      return "compile-prompt";

    case "model-route":
      return "choose-models";

    case "planning":
    case "orchestration":
      return "reason-plan";

    case "verification":
      return failedStage ===
        "verify"
        ? "verify"
        : "validate";

    case "authority":
      return "authorize";

    case "action":
      return "act";

    case "human":
    case "stop":
      return undefined;
  }
}

function recordArtifact(
  state:
    AutonomousCognitiveLoopState,
  stage:
    AutonomousCognitiveStage,
  artifact:
    unknown,
): void {
  if (
    artifact ===
    undefined
  ) {
    return;
  }

  const safe =
    jsonSafe(
      artifact,
    );

  if (
    state.artifacts[
      stage
    ] !==
    undefined
  ) {
    const history =
      state
        .supersededArtifacts[
          stage
        ] ??
      [];

    history.push(
      state.artifacts[
        stage
      ],
    );

    state.supersededArtifacts[
      stage
    ] =
      history;
  }

  state.artifacts[
    stage
  ] =
    safe;
}

function resetFromStage(
  state:
    AutonomousCognitiveLoopState,
  resumeStage:
    AutonomousCognitiveStage,
): void {
  const start =
    stageIndex(
      resumeStage,
    );

  for (
    let index =
      start;
    index <
      state.stages.length;
    index +=
      1
  ) {
    const stageState =
      state.stages[
        index
      ];

    if (
      !stageState
    ) {
      continue;
    }

    if (
      state.artifacts[
        stageState.stage
      ] !==
      undefined
    ) {
      const history =
        state
          .supersededArtifacts[
            stageState.stage
          ] ??
        [];

      history.push(
        state.artifacts[
          stageState.stage
        ],
      );

      state
        .supersededArtifacts[
          stageState.stage
        ] =
        history;

      delete state
        .artifacts[
          stageState.stage
        ];
    }

    stageState.status =
      "pending";

    stageState.startedAt =
      undefined;

    stageState.completedAt =
      undefined;

    stageState.error =
      undefined;
  }

  state.currentStageIndex =
    start;
}

function hasPassedAct(
  state:
    AutonomousCognitiveLoopState,
): boolean {
  const act =
    state.stages[
      stageIndex(
        "act",
      )
    ];

  return (
    act?.status ===
      "completed" ||
    act?.status ===
      "skipped"
  );
}

function safePostActionRecovery(
  decision:
    FailureRecoveryDecision,
  state:
    AutonomousCognitiveLoopState,
  failedStage:
    AutonomousCognitiveStage,
  recoveryContext:
    Partial<
      FailureRecoveryContext
    >,
): boolean {
  if (
    !hasPassedAct(
      state,
    ) &&
    stageIndex(
      failedStage,
    ) <=
      stageIndex(
        "act",
      )
  ) {
    return true;
  }

  if (
    decision.selected
      .action ===
        "retry-tool" ||
    decision.selected
      .action ===
        "resume-from-checkpoint"
  ) {
    const partial =
      recoveryContext
        .partialExecution;

    return Boolean(
      partial
        ?.resumable &&
      partial
        .checkpointRef &&
      (
        !partial
          .sideEffectsObserved ||
        partial
          .idempotencyKey
      ),
    );
  }

  if (
    decision.selected
      .action ===
      "invoke-critic" &&
    failedStage ===
      "verify"
  ) {
    return true;
  }

  return (
    decision.selected
      .resumeStage ===
      "human" ||
    decision.selected
      .resumeStage ===
      "stop"
  );
}

export class JsonAutonomousCognitiveLoopStore
  implements AutonomousCognitiveLoopStore {
  constructor(
    private readonly directory =
      join(
        process.cwd(),
        ".chernobog",
        "cognition",
        "autonomous-loop",
      ),
  ) {}

  async save(
    state:
      AutonomousCognitiveLoopState,
  ): Promise<void> {
    const safe =
      jsonSafe(
        state,
      );

    const target =
      this.pathFor(
        state.id,
      );

    await mkdir(
      dirname(
        target,
      ),
      {
        recursive:
          true,
      },
    );

    const temporary =
      `${target}.${process.pid}.tmp`;

    await writeFile(
      temporary,
      `${JSON.stringify(safe, null, 2)}\n`,
      "utf8",
    );

    await rename(
      temporary,
      target,
    );
  }

  async load(
    id: string,
  ): Promise<
    AutonomousCognitiveLoopState
  > {
    const raw =
      await readFile(
        this.pathFor(
          id,
        ),
        "utf8",
      );

    const parsed =
      JSON.parse(
        raw,
      ) as
        AutonomousCognitiveLoopState;

    if (
      parsed.version !==
      AUTONOMOUS_COGNITIVE_LOOP_VERSION
    ) {
      throw new Error(
        `Unsupported OH-4AD loop state version: ${String(parsed.version)}`,
      );
    }

    return cloneState(
      parsed,
    );
  }

  private pathFor(
    id: string,
  ): string {
    if (
      !/^oh4ad-[a-zA-Z0-9-]+$/
        .test(
          id,
        )
    ) {
      throw new Error(
        "Invalid OH-4AD loop id.",
      );
    }

    return join(
      this.directory,
      `${id}.json`,
    );
  }
}

async function defaultEventPublisher(
  event:
    AutonomousCognitiveLoopEvent,
  state:
    AutonomousCognitiveLoopState,
): Promise<void> {
  await publishChernobogEventSafely({
    type:
      event.type,

    source: {
      subsystem:
        "cognition.autonomous-loop",
    },

    severity:
      event.status ===
        "failed" ||
      event.status ===
        "stopped"
        ? "warning"
        : event.status ===
            "waiting-human" ||
          event.status ===
            "interrupted"
          ? "notice"
          : "info",

    subject:
      state.input
        .subject ??
      state.id,

    scope:
      state.input
        .scope ??
      "cognition:oh4ad",

    correlationId:
      state.input
        .correlationId ??
      state.id,

    causationId:
      event.stage
        ? `${state.id}:${event.stage}`
        : undefined,

    payload: {
      loopId:
        state.id,

      stage:
        event.stage,

      status:
        event.status,

      detail:
        event.detail,

      currentStageIndex:
        state.currentStageIndex,

      totalStageExecutions:
        state
          .totalStageExecutions,

      recoveryCount:
        state.recoveries
          .length,
    },

    metadata: {
      tags: [
        "oh4",
        "oh4ad",
        "autonomous-cognitive-loop",
        ...(
          event.stage
            ? [
                event.stage,
              ]
            : []
        ),
      ],
    },
  });
}

export class AutonomousCognitiveLoop {

  private readonly store:

  private readonly clock:

  constructor(

  async run(
    id: string,
    options:
      AutonomousCognitiveRunOptions =
      {},
  ): Promise<
    AutonomousCognitiveLoopState
  > {
    let state =
      await this.store
        .load(
          id,
        );

    if (
      isTerminal(
        state,
      )
    ) {
      return state;
    }

    if (
      state.status ===
        "waiting-human"
    ) {
      return state;
    }

    if (
      state.status ===
        "interrupted"
    ) {
      return state;
    }

    state.status =
      "running";

    state.updatedAt =
      nowIso(
        this.clock,
      );

    await this.persist(
      state,
    );

    while (
      state.currentStageIndex <
      AUTONOMOUS_COGNITIVE_STAGE_ORDER
        .length
    ) {
      if (
        state.totalStageExecutions >=
        this.policy
          .maxStageExecutions
      ) {
        state.status =
          "stopped";

        state.terminalReason =
          `OH-4AD exceeded max stage execution limit of ${this.policy.maxStageExecutions}.`;

        await this.persist(
          state,
        );

        await this.emit(
          {
            type:
              "cognition.autonomous-loop.stopped",

            loopId:
              state.id,

            status:
              "stopped",

            detail:
              state
                .terminalReason,
          },
          state,
        );

        return cloneState(
          state,
        );
      }

      if (
        await options
          .shouldInterrupt?.(
            cloneState(
              state,
            ),
          )
      ) {
        return this.interrupt(
          state.id,
          "External interruption requested.",
        );
      }

      const stage =
        AUTONOMOUS_COGNITIVE_STAGE_ORDER[
          state.currentStageIndex
        ];

      if (!stage) {
        break;
      }

      const stageState =
        state.stages[
          state.currentStageIndex
        ];

      if (!stageState) {
        throw new Error(
          `Missing OH-4AD stage state at index ${state.currentStageIndex}.`,
        );
      }

      if (
        stageState.status ===
          "completed" ||
        stageState.status ===
          "skipped"
      ) {
        state.currentStageIndex +=
          1;

        continue;
      }

      stageState.status =
        "running";

      stageState.attempts +=
        1;

      stageState.startedAt =
        nowIso(
          this.clock,
        );

      stageState.error =
        undefined;

      state.totalStageExecutions +=
        1;

      state.updatedAt =
        stageState.startedAt;

      await this.persist(
        state,
      );

      await this.emit(
        {
          type:
            "cognition.autonomous-loop.stage.started",

          loopId:
            state.id,

          stage,

          status:
            "running",
        },
        state,
      );

      const controller =
        new AbortController();

      const humanInput =
        state
          .humanInterventions
          .filter(
            (entry) =>
              entry.stage ===
              stage,
          )
          .at(
            -1,
          )
          ?.input;

      let result:
        AutonomousCognitiveStageResult;

      try {
        result =
          await this.handlers[
            stage
          ]({
            loopId:
              state.id,

            stage,

            input:
              structuredClone(
                state.input,
              ),

            artifacts:
              structuredClone(
                state.artifacts,
              ),

            stageStates:
              structuredClone(
                state.stages,
              ),

            humanInput:
              humanInput ===
                undefined
                ? undefined
                : structuredClone(
                    humanInput,
                  ),

            recoveryHistory:
              structuredClone(
                state.recoveries,
              ),

            signal:
              controller.signal,
          });
      } catch (error) {
        const message =
          error instanceof Error
            ? error.message
            : String(
                error,
              );

        result = {
          status:
            "failed",

          error:
            message,

          failure: {
            kind:
              "unknown",

            message,
          },
        };
      }

      recordArtifact(
        state,
        stage,
        result.artifact,
      );

      if (
        result.status ===
          "completed" ||
        result.status ===
          "skipped"
      ) {
        stageState.status =
          result.status;

        stageState.completedAt =
          nowIso(
            this.clock,
          );

        stageState.notes = [
          ...stageState
            .notes,
          ...normalizeNotes(
            result.notes,
          ),
        ];

        state.currentStageIndex +=
          1;

        state.updatedAt =
          stageState
            .completedAt;

        await this.persist(
          state,
        );

        await this.emit(
          {
            type:
              "cognition.autonomous-loop.stage.completed",

            loopId:
              state.id,

            stage,

            status:
              result.status,
          },
          state,
        );

        continue;
      }

      if (
        result.status ===
        "waiting-human"
      ) {
        const questions =
          normalizeQuestions(
            result.questions,
          );

        stageState.status =
          "waiting-human";

        stageState.error =
          undefined;

        state.status =
          "waiting-human";

        state.waitingForHuman = {
          stage,

          reason:
            result.reason
              .trim() ||
            "Human intervention is required.",

          questions,

          recoveryDriven:
            false,
        };

        state.updatedAt =
          nowIso(
            this.clock,
          );

        await this.persist(
          state,
        );

        await this.emit(
          {
            type:
              "cognition.autonomous-loop.waiting-human",

            loopId:
              state.id,

            stage,

            status:
              "waiting-human",

            detail:
              state
                .waitingForHuman
                .reason,
          },
          state,
        );

        return cloneState(
          state,
        );
      }

      if (
        result.status !==
        "failed"
      ) {
        throw new Error(
          `OH-4AD stage "${stage}" returned an unsupported result status.`,
        );
      }

      state =
        await this.handleFailure(
          state,
          stage,
          result,
        );

      if (
        state.status ===
          "waiting-human" ||
        state.status ===
          "stopped" ||
        state.status ===
          "failed"
      ) {
        return cloneState(
          state,
        );
      }
    }

    state.status =
      "completed";

    state.currentStageIndex =
      AUTONOMOUS_COGNITIVE_STAGE_ORDER
        .length;

    state.updatedAt =
      nowIso(
        this.clock,
      );

    await this.persist(
      state,
    );

    await this.emit(
      {
        type:
          "cognition.autonomous-loop.completed",

        loopId:
          state.id,

        status:
          "completed",
      },
      state,
    );

    return cloneState(
      state,
    );
  }

    const state =
      await this.store

    if (
      isTerminal(
        state,

      return state;

      state.stages[
        stageIndex(
          stage,

      ];

    if (
      stageState

        "running"
    ) {
      stageState.status =
        "pending";
    }

    await this.persist(
      state,
    );

    await this.emit(
      {
        type:
          "cognition.autonomous-loop.interrupted",

    );
  }

  async resume(
    id: string,
    options:
      AutonomousCognitiveResumeOptions =
      {},
  ): Promise<
    AutonomousCognitiveLoopState
  > {
    const state =
      await this.store
        .load(
          id,
        );

    if (
      isTerminal(
        state,
      )
    ) {
      return state;
    }

    if (
      state.status ===
        "waiting-human"
    ) {
      if (
        options.humanInput ===
        undefined
      ) {
        throw new Error(
          "OH-4AD loop is waiting for human intervention; humanInput is required to resume.",
        );
      }

      const stage =
        state
          .waitingForHuman
          ?.stage ??
        currentStage(
          state,
        );

      state
        .humanInterventions
        .push({
          at:
            nowIso(
              this.clock,
            ),

          stage,

          input:
            jsonSafe(
              options.humanInput,
            ),
        });

      const stageState =
        state.stages[
          stageIndex(
            stage,
          )
        ];

      if (
        stageState
      ) {
        stageState.status =
          "pending";

        stageState.error =
          undefined;
      }

      state.waitingForHuman =
        undefined;
    }

    if (
      state.status ===
        "interrupted"
    ) {
      const stage =
        state
          .interruption
          ?.stage ??
        currentStage(
          state,
        );

      const stageState =
        state.stages[
          stageIndex(
            stage,
          )
        ];

      if (
        stageState &&
        stageState.status !==
          "completed" &&
        stageState.status !==
          "skipped"
      ) {
        stageState.status =
          "pending";
      }

      state.interruption =
        undefined;
    }

    state.status =
      "running";

    state.updatedAt =
      nowIso(
        this.clock,
      );

    await this.persist(
      state,
    );

    return this.run(
      id,
      options,
    );
  }

  private async handleFailure(
    state:
      AutonomousCognitiveLoopState,
    failedStage:
      AutonomousCognitiveStage,
    result:
      Extract<
        AutonomousCognitiveStageResult,
        {
          status:
            "failed";
        }
      >,
  ): Promise<
    AutonomousCognitiveLoopState
  > {
    const failedState =
      state.stages[
        stageIndex(
          failedStage,
        )
      ];

    if (
      failedState
    ) {
      failedState.status =
        "failed";

      failedState.error =
        result.error;

      failedState.completedAt =
        nowIso(
          this.clock,
        );
    }

    await this.emit(
      {
        type:
          "cognition.autonomous-loop.stage.failed",

        loopId:
          state.id,

        stage:
          failedStage,

        status:
          "failed",

        detail:
          result.error,
      },
      state,
    );

    if (
      state.recoveries
        .length >=
      this.policy
        .maxRecoveryAttempts
    ) {
      state.status =
        "stopped";

      state.terminalReason =
        `OH-4AD exceeded max recovery attempt limit of ${this.policy.maxRecoveryAttempts}.`;

      await this.persist(
        state,
      );

      return state;
    }

    const recoveryContext:
      FailureRecoveryContext = {
        attemptCount:
          state.recoveries
            .length,

        maxAttempts:
          this.policy
            .maxRecoveryAttempts,

        recentAttempts:
          recoveryAttemptsForEngine(
            state,
          ),

        humanAvailable:
          true,

        ...(
          result.recoveryContext ??
          {}
        ),
      };

    const failure:
      FailureSignal = {
        id:
          result.failure
            ?.id ??
          `${state.id}:${failedStage}:${failedState?.attempts ?? 1}`,

        kind:
          result.failure
            ?.kind ??
          "unknown",

        occurredAt:
          result.failure
            ?.occurredAt ??
          nowIso(
            this.clock,
          ),

        stage:
          result.failure
            ?.stage ??
          failedStage,

        source:
          result.failure
            ?.source ??
          "oh4ad",

        message:
          result.failure
            ?.message ??
          result.error,

        transient:
          result.failure
            ?.transient,

        retryable:
          result.failure
            ?.retryable,

        failureFingerprint:
          result.failure
            ?.failureFingerprint,

        modelId:
          result.failure
            ?.modelId,

        toolName:
          result.failure
            ?.toolName,

        evidenceRefs:
          result.failure
            ?.evidenceRefs,
      };

    const decision =
      getFailureRecoveryEngine()
        .decide({
          failure,

          context:
            recoveryContext,

          policy: {
            maxAttempts:
              this.policy
                .maxRecoveryAttempts,
          },
        });

    const safe =
      safePostActionRecovery(
        decision,
        state,
        failedStage,
        recoveryContext,
      );

    if (!safe) {
      state.status =
        "waiting-human";

      state.waitingForHuman = {
        stage:
          failedStage,

        reason:
          "Automatic recovery would rewind across a completed action without explicit safe replay/resume evidence.",

        questions: [
          "An external or side-effecting action may already have occurred. Review the current state and choose whether to continue, repair manually, or stop.",
        ],

        recoveryDriven:
          true,
      };

      state.updatedAt =
        nowIso(
          this.clock,
        );

      await this.persist(
        state,
      );

      await this.emit(
        {
          type:
            "cognition.autonomous-loop.waiting-human",

          loopId:
            state.id,

          stage:
            failedStage,

          status:
            "waiting-human",

          detail:
            state
              .waitingForHuman
              .reason,
        },
        state,
      );

      return state;
    }

    const mappedStage =
      recoveryResumeToStage(
        decision.selected
          .resumeStage,
        decision.selected
          .action,
        failedStage,
      );

    state.recoveries
      .push({
        at:
          nowIso(
            this.clock,
          ),

        failedStage,

        failureFingerprint:
          decision.failure
            .fingerprint,

        action:
          decision.selected
            .action,

        resumeStage:
          decision.selected
            .resumeStage,

        resumedAtStage:
          mappedStage,

        summary:
          decision.selected
            .summary,
      });

    if (
      decision.selected
        .action ===
        "request-human-clarification"
    ) {
      state.status =
        "waiting-human";

      state.waitingForHuman = {
        stage:
          failedStage,

        reason:
          decision.selected
            .summary,

        questions: [
          decision.selected
            .summary,
        ],

        recoveryDriven:
          true,
      };

      state.updatedAt =
        nowIso(
          this.clock,
        );

      await this.persist(
        state,
      );

      return state;
    }

    if (
      decision.selected
        .action ===
        "stop-safely" ||
      !mappedStage
    ) {
      state.status =
        "stopped";

      state.terminalReason =
        decision.selected
          .summary;

      state.updatedAt =
        nowIso(
          this.clock,
        );

      await this.persist(
        state,
      );

      return state;
    }

    resetFromStage(
      state,
      mappedStage,
    );

    state.status =
      "recovering";

    state.updatedAt =
      nowIso(
        this.clock,
      );

    await this.persist(
      state,
    );

    await this.emit(
      {
        type:
          "cognition.autonomous-loop.recovering",

        loopId:
          state.id,

        stage:
          mappedStage,

        status:
          "recovering",

        detail:
          `${decision.selected.action}: ${decision.selected.summary}`,
      },
      state,
    );

    state.status =
      "running";

    state.updatedAt =
      nowIso(
        this.clock,
      );

    await this.persist(
      state,
    );

    return state;
  }

  private async persist(
    state:
      AutonomousCognitiveLoopState,
  ): Promise<void> {
    state.updatedAt =
      nowIso(
        this.clock,
      );

    await this.store
      .save(
        state,
      );
  }

  private async emit(
    event:
      AutonomousCognitiveLoopEvent,
    state:
      AutonomousCognitiveLoopState,
  ): Promise<void> {
    await this.eventPublisher(
      event,
      cloneState(
        state,
      ),
    );
  }
}

): string {
  return typeof value ===
    "string"
    ? value
    : fallback;
}

function asBoolean(
  value: unknown,
  fallback = false,
): boolean {
  return typeof value ===
    "boolean"
    ? value
    : fallback;
}

function asNumber(
  value: unknown,
): number | undefined {
  return (
    typeof value ===
      "number" &&
    Number.isFinite(
      value,
    )
  )
    ? value
    : undefined;
}

function asStrings(
  value: unknown,
): string[] {
  if (
    !Array.isArray(
      value,
    )
  ) {
    return [];
  }

  return [
    ...new Set(
      value
        .filter(
          (
            entry,
          ): entry is string =>
            typeof entry ===
            "string",
        )
        .map(
          (entry) =>
            entry
              .trim()
              .toLowerCase(),
        )
        .filter(Boolean),
    ),
  ];
}

function normalizeRegistryCandidate(
  value: unknown,
): ModelRoutingCandidate | null {
  const record =
    asRecord(
      value,
    );

  const provider =
    asString(
      record.provider,
    ).trim();

  const model =
    asString(
      record.model,
    ).trim();

  const id =
    asString(
      record.id,
      provider &&
      model
        ? `${provider}:${model}`
        : "",
    ).trim();

  if (
    !id ||
    !provider ||
    !model
  ) {
    return null;
  }

  const contextLimits =
    asRecord(
      record.contextLimits,
    );

  const cost =
    asRecord(
      record.cost,
    );

  const privacy =
    asRecord(
      record.privacy,
    );

  const availabilityRaw =
    asString(
      record.availability,
      "unknown",
    );

  const availability =
    availabilityRaw ===
      "available" ||
    availabilityRaw ===
      "unavailable"
      ? availabilityRaw
      : "unknown";

  const costModeRaw =
    asString(
      cost.mode,
      "unknown",
    );

  const costMode:
    ModelRoutingCandidate[
      "cost"
    ]["mode"] =
      costModeRaw ===
        "local-compute" ||
      costModeRaw ===
        "metered" ||
      costModeRaw ===
        "subscription"
        ? costModeRaw
        : "unknown";

  const executionRaw =
    asString(
      privacy.execution,
      "unknown",
    );

  const execution:
    ModelRoutingCandidate[
      "privacy"
    ]["execution"] =
      executionRaw ===
        "local" ||
      executionRaw ===
        "remote" ||
      executionRaw ===
        "hybrid"
        ? executionRaw
        : "unknown";

  const retentionRaw =
    asString(
      privacy.retention,
      "unknown",
    );

  const retention:
    ModelRoutingCandidate[
      "privacy"
    ]["retention"] =
      retentionRaw ===
        "none" ||
      retentionRaw ===
        "local-runtime" ||
      retentionRaw ===
        "provider-policy"
        ? retentionRaw
        : "unknown";

  const promptLeavesDevice =
    typeof privacy
      .promptLeavesDevice ===
      "boolean"
      ? privacy
          .promptLeavesDevice
      : null;

  return {
    id,
    provider,
    model,
    displayName:
      asString(
        record.displayName,
        id,
      ),
    enabled:
      asBoolean(
        record.enabled,
        true,
      ),
    availability,
    local:
      asBoolean(
        record.local,
        execution ===
          "local",
      ),
    roles:
      asStrings(
        record.roles,
      ),
    capabilities:
      asStrings(
        record.capabilities,
      ),
    contextLimits: {
      known:
        asBoolean(
          contextLimits.known,
          false,
        ),
      contextWindowTokens:
        asNumber(
          contextLimits
            .contextWindowTokens,
        ),
      maxOutputTokens:
        asNumber(
          contextLimits
            .maxOutputTokens,
        ),
    },
    cost: {
      known:
        asBoolean(
          cost.known,
          false,
        ),
      mode:
        costMode,
      currency:
        asString(
          cost.currency,
        ) ||
        undefined,
      inputPerMillionTokens:
        asNumber(
          cost
            .inputPerMillionTokens,
        ),
      outputPerMillionTokens:
        asNumber(
          cost
            .outputPerMillionTokens,
        ),
    },
    privacy: {
      execution,
      promptLeavesDevice,
      retention,
    },
    strengths:
      asStrings(
        record.strengths,
      ),
    weaknesses:
      asStrings(
        record.weaknesses,
      ),
    tags:
      asStrings(
        record.tags,
      ),
  };
}

function defaultCandidateSource():
  ModelRoutingCandidate[] {
  const registry =
    getDefaultModelRegistry();

  return registry
    .list()
    .map(
      (entry) =>
        normalizeRegistryCandidate(
          entry,
        ),
    )
    .filter(
      (
        entry,
      ): entry is
        ModelRoutingCandidate =>
          entry !== null,
    );
}

function normalizedSet(
  values:
    readonly string[]
    | undefined,
): Set<string> {
  return new Set(
    (values ?? [])
      .map(
        (value) =>
          value
            .trim()
            .toLowerCase(),
      )
      .filter(Boolean),
  );
}

function taskTerms(
  interpretation:
    InputInterpretation,
): Set<string> {
  return normalizedSet([
    interpretation
      .taskIntelligence
      .primaryDomain.value,
    ...interpretation
      .taskIntelligence
      .secondaryDomains,
    ...(
      interpretation
        .taskSignals
        .requiredTools ??
      []
    ),
  ]);
}

function candidateTerms(
  candidate:
    ModelRoutingCandidate,
): Set<string> {
  return normalizedSet([
    ...candidate.roles,
    ...candidate.capabilities,
    ...candidate.strengths,
    ...candidate.tags,
  ]);
}

function taskFitScore(
  candidate:
    ModelRoutingCandidate,
  interpretation:
    InputInterpretation,
): {
  score: number;
  reasons: string[];
} {
  const wanted =
    taskTerms(
      interpretation,
    );

  const offered =
    candidateTerms(
      candidate,
    );

  if (
    wanted.size ===
    0
  ) {
    return {
      score:
        0.5,
      reasons: [
        "No task-fit terms were available; neutral task-fit score used.",
      ],
    };
  }

  let matches =
    0;

  const matched:
    string[] =
      [];

  for (
    const term
    of wanted
  ) {
    if (
      offered.has(
        term,
      )
    ) {
      matches += 1;
      matched.push(
        term,
      );
    }
  }

  const domain =
    interpretation
      .taskIntelligence
      .primaryDomain.value
      .toLowerCase();

  let domainBoost =
    0;

  if (
    offered.has(
      domain,
    )
  ) {
    domainBoost =
      0.25;
  }

  const score =
    clamp01(
      0.35 +
      Math.min(
        0.4,
        matches *
          0.12,
      ) +
      domainBoost,
    );

  return {
    score,
    reasons:
      matched.length >
      0
        ? [
            `Matched task-fit terms: ${matched.join(", ")}.`,
          ]
        : [
            "No declared capability/tag directly matched the task; conservative baseline used.",
          ],
  };
}

function privacyScore(
  candidate:
    ModelRoutingCandidate,
  interpretation:
    InputInterpretation,
): {
  score: number;
  reasons: string[];
} {
  const privacy =
    interpretation
      .taskIntelligence
      .privacySensitivity.value;

  const local =
    candidate.local ||
    candidate.privacy
      .execution ===
      "local";

  const leaves =
    candidate.privacy
      .promptLeavesDevice ===
      true;

  if (
    privacy ===
      "highly-sensitive"
  ) {
    if (
      local &&
      !leaves
    ) {
      return {
        score:
          1,
        reasons: [
          "Highly-sensitive task remains local.",
        ],
      };
    }

    return {
      score:
        0,
      reasons: [
        "Highly-sensitive task is not guaranteed to remain local.",
      ],
    };
  }

  if (
    privacy ===
      "sensitive"
  ) {
    return local &&
      !leaves
      ? {
          score:
            1,
          reasons: [
            "Sensitive task benefits from local execution.",
          ],
        }
      : {
          score:
            0.35,
          reasons: [
            "Sensitive task would use non-local/uncertain privacy handling.",
          ],
        };
  }

  if (
    privacy ===
      "personal" ||
    privacy ===
      "internal"
  ) {
    return local
      ? {
          score:
            0.9,
          reasons: [
            "Local execution reduces exposure for non-public context.",
          ],
        }
      : {
          score:
            0.65,
          reasons: [
            "Remote execution is acceptable but less privacy-preserving than local.",
          ],
        };
  }

  return {
    score:
      local
        ? 0.85
        : 0.75,
    reasons: [
      "Public/low-sensitivity task has no strong privacy restriction.",
    ],
  };
}

function costScore(
  candidate:
    ModelRoutingCandidate,
  interpretation:
    InputInterpretation,
  constraints:
    ModelRoutingConstraints,
): {
  score: number;
  reasons: string[];
} {
  if (
    candidate.cost.mode ===
      "local-compute"
  ) {
    return {
      score:
        1,
      reasons: [
        "Local-compute model has no metered token charge in the registry.",
      ],
    };
  }

  const inputCost =
    candidate.cost
      .inputPerMillionTokens;

  const outputCost =
    candidate.cost
      .outputPerMillionTokens;

  if (
    constraints
      .maxInputCostPerMillion !==
      undefined &&
    inputCost !==
      undefined &&
    inputCost >
      constraints
        .maxInputCostPerMillion
  ) {
    return {
      score:
        0,
      reasons: [
        "Input token cost exceeds explicit routing ceiling.",
      ],
    };
  }

  if (
    constraints
      .maxOutputCostPerMillion !==
      undefined &&
    outputCost !==
      undefined &&
    outputCost >
      constraints
        .maxOutputCostPerMillion
  ) {
    return {
      score:
        0,
      reasons: [
        "Output token cost exceeds explicit routing ceiling.",
      ],
    };
  }

  const tolerance =
    interpretation
      .taskIntelligence
      .costTolerance.value;

  if (
    !candidate.cost.known
  ) {
    return {
      score:
        tolerance ===
          "minimal"
          ? 0.35
          : 0.5,
      reasons: [
        "Cost is unknown; router does not assume it is cheap.",
      ],
    };
  }

  const combined =
    (
      inputCost ??
      0
    ) +
    (
      outputCost ??
      0
    );

  const divisor =
    tolerance ===
      "minimal"
      ? 10
      : tolerance ===
          "low"
        ? 25
        : tolerance ===
            "standard"
          ? 75
          : 150;

  return {
    score:
      clamp01(
        1 -
        combined /
          divisor,
      ),
    reasons: [
      `Known metered cost considered against ${tolerance} task tolerance.`,
    ],
  };
}

function latencyScore(
  candidate:
    ModelRoutingCandidate,
  interpretation:
    InputInterpretation,
  evidence:
    ModelRoutingEvidence,
  constraints:
    ModelRoutingConstraints,
): {
  score: number;
  confidence: number;
  measured: boolean;
  reasons: string[];
} {
  const telemetry =
    evidence.telemetry
      ?.[candidate.id];

  const latency =
    telemetry
      ?.averageLatencyMs;

  if (
    latency ===
    undefined
  ) {
    return {
      score:
        interpretation
          .taskIntelligence
          .latencyTolerance.value ===
          "immediate"
          ? 0.4
          : 0.55,
      confidence:
        0.25,
      measured:
        false,
      reasons: [
        "No measured latency evidence is available.",
      ],
    };
  }

  if (
    constraints
      .maxLatencyMs !==
      undefined &&
    latency >
      constraints
        .maxLatencyMs
  ) {
    return {
      score:
        0,
      confidence:
        clamp01(
          telemetry
            ?.confidence ??
          0.7,
        ),
      measured:
        true,
      reasons: [
        `Measured latency ${latency}ms exceeds explicit ceiling.`,
      ],
    };
  }

  const tolerance =
    interpretation
      .taskIntelligence
      .latencyTolerance.value;

  const target =
    tolerance ===
      "immediate"
      ? 1_500
      : tolerance ===
          "interactive"
        ? 5_000
        : tolerance ===
            "flexible"
          ? 20_000
          : 60_000;

  return {
    score:
      clamp01(
        target /
        Math.max(
          target,
          latency,
        ),
      ),
    confidence:
      clamp01(
        telemetry
          ?.confidence ??
        0.7,
      ),
    measured:
      true,
    reasons: [
      `Measured average latency ${latency}ms evaluated against ${tolerance} target ${target}ms.`,
    ],
  };
}

function reliabilityScore(
  candidate:
    ModelRoutingCandidate,
  evidence:
    ModelRoutingEvidence,
): {
  score: number;
  confidence: number;
  measured: boolean;
  reasons: string[];
} {
  const telemetry =
    evidence.telemetry
      ?.[candidate.id];

  if (
    !telemetry
  ) {
    return {
      score:
        0.5,
      confidence:
        0.2,
      measured:
        false,
      reasons: [
        "No measured reliability evidence is available.",
      ],
    };
  }

  const success =
    telemetry.successRate;

  const failure =
    telemetry.failureRate;

  if (
    success ===
      undefined &&
    failure ===
      undefined
  ) {
    return {
      score:
        0.5,
      confidence:
        clamp01(
          telemetry.confidence ??
          0.3,
        ),
      measured:
        false,
      reasons: [
        "Telemetry exists but contains no success/failure rate.",
      ],
    };
  }

  const score =
    success !==
      undefined
      ? clamp01(
          success,
        )
      : clamp01(
          1 -
          (
            failure ??
            0
          ),
        );

  return {
    score,
    confidence:
      clamp01(
        telemetry.confidence ??
        0.7,
      ),
    measured:
      true,
    reasons: [
      "Reliability score derives from measured invocation telemetry.",
    ],
  };
}

function availabilityScore(
  candidate:
    ModelRoutingCandidate,
): number {
  switch (
    candidate.availability
  ) {
    case "available":
      return 1;
    case "unknown":
      return 0.5;
    case "unavailable":
    default:
      return 0;
  }
}

function dynamicWeights(
  interpretation:
    InputInterpretation,
): Record<
  ModelRouteComponent,
  number
> {
  const weights:
    Record<
      ModelRouteComponent,
      number
    > = {
      "objective-calibration":
        0.24,
      "task-fit":
        0.18,
      privacy:
        0.14,
      latency:
        0.1,
      cost:
        0.09,
      availability:
        0.08,
      "subjective-preference":
        0.07,
      reliability:
        0.1,
    };

  const factuality =
    interpretation
      .taskIntelligence
      .factuality.value;

  const privacy =
    interpretation
      .taskIntelligence
      .privacySensitivity.value;

  const latency =
    interpretation
      .taskIntelligence
      .latencyTolerance.value;

  const cost =
    interpretation
      .taskIntelligence
      .costTolerance.value;

  if (
    factuality ===
      "critical" ||
    factuality ===
      "high"
  ) {
    weights[
      "objective-calibration"
    ] += 0.08;
    weights[
      "subjective-preference"
    ] -= 0.03;
    weights.cost -=
      0.02;
    weights.latency -=
      0.03;
  }

  if (
    privacy ===
      "highly-sensitive" ||
    privacy ===
      "sensitive"
  ) {
    weights.privacy +=
      0.1;
    weights[
      "subjective-preference"
    ] -= 0.02;
    weights.cost -=
      0.03;
    weights.latency -=
      0.05;
  }

  if (
    latency ===
      "immediate"
  ) {
    weights.latency +=
      0.1;
    weights[
      "objective-calibration"
    ] -= 0.04;
    weights[
      "subjective-preference"
    ] -= 0.02;
    weights.cost -=
      0.04;
  }

  if (
    cost ===
      "minimal" ||
    cost ===
      "low"
  ) {
    weights.cost +=
      0.1;
    weights[
      "subjective-preference"
    ] -= 0.02;
    weights[
      "objective-calibration"
    ] -= 0.04;
    weights.latency -=
      0.04;
  }

  const total =
    Object.values(
      weights,
    ).reduce(
      (
        sum,
        value,
      ) =>
        sum +
        Math.max(
          0,
          value,
        ),
      0,
    );

  for (
    const key
    of Object.keys(
      weights,
    ) as
      ModelRouteComponent[]
  ) {
    weights[key] =
      total > 0
        ? Math.max(
            0,
            weights[key],
          ) /
          total
        : 0;
  }

  return weights;
}

function eligibility(
  candidate:
    ModelRoutingCandidate,
  interpretation:
    InputInterpretation,
  constraints:
    ModelRoutingConstraints,
): string[] {
  const reasons:
    string[] =
      [];

  const required =
    normalizedSet(
      constraints
        .requiredModelIds,
    );

  const excluded =
    normalizedSet(
      constraints
        .excludedModelIds,
    );

  if (
    !candidate.enabled
  ) {
    reasons.push(
      "model is disabled",
    );
  }

  if (
    candidate.availability ===
      "unavailable"
  ) {
    reasons.push(
      "model is unavailable",
    );
  }

  if (
    required.size >
      0 &&
    !required.has(
      candidate.id
        .toLowerCase(),
    )
  ) {
    reasons.push(
      "model is not in requiredModelIds",
    );
  }

  if (
    excluded.has(
      candidate.id
        .toLowerCase(),
    )
  ) {
    reasons.push(
      "model is explicitly excluded",
    );
  }

  const isLocal =
    candidate.local ||
    candidate.privacy
      .execution ===
      "local";

  if (
    constraints.requireLocal &&
    !isLocal
  ) {
    reasons.push(
      "explicit constraint requires local execution",
    );
  }

  if (
    constraints.allowRemote ===
      false &&
    !isLocal
  ) {
    reasons.push(
      "remote execution is explicitly disallowed",
    );
  }

  const privacySensitivity =
    interpretation
      .taskIntelligence
      .privacySensitivity
      .value;

  const privacyRequiresLocal =
    privacySensitivity ===
      "highly-sensitive" ||
    interpretation
      .taskIntelligence
      .derived
      .prefersLocalExecution ===
      true;

  if (
    privacyRequiresLocal &&
    (
      !isLocal ||
      candidate.privacy
        .promptLeavesDevice ===
        true
    )
  ) {
    reasons.push(
      privacySensitivity ===
        "highly-sensitive"
        ? "highly-sensitive task requires a model that keeps the prompt local"
        : "task intelligence requires local execution for this privacy-sensitive task",
    );
  }

  if (
    constraints
      .maxInputCostPerMillion !==
      undefined &&
    candidate.cost
      .inputPerMillionTokens !==
      undefined &&
    candidate.cost
      .inputPerMillionTokens >
      constraints
        .maxInputCostPerMillion
  ) {
    reasons.push(
      "input cost exceeds explicit ceiling",
    );
  }

  if (
    constraints
      .maxOutputCostPerMillion !==
      undefined &&
    candidate.cost
      .outputPerMillionTokens !==
      undefined &&
    candidate.cost
      .outputPerMillionTokens >
      constraints
        .maxOutputCostPerMillion
  ) {
    reasons.push(
      "output cost exceeds explicit ceiling",
    );
  }

  return reasons;
}

function component(
  name:
    ModelRouteComponent,
  score:
    number,
  weight:
    number,
  confidence:
    number,
  evidence:
    ModelRouteScoreComponent[
      "evidence"
    ],
  reasons:
    string[],
): ModelRouteScoreComponent {
  const normalizedScore =
    clamp01(
      score,
    );

  const normalizedWeight =
    Math.max(
      0,
      weight,
    );

  return {
    component:
      name,
    score:
      normalizedScore,
    weight:
      normalizedWeight,
    weightedScore:
      normalizedScore *
      normalizedWeight,
    confidence:
      clamp01(
        confidence,
      ),
    evidence,
    reasons,
  };
}

function scoreCandidate(
  candidate:
    ModelRoutingCandidate,
  interpretation:
    InputInterpretation,
  evidence:
    ModelRoutingEvidence,
  constraints:
    ModelRoutingConstraints,
): RankedModelCandidate {
  const exclusions =
    eligibility(
      candidate,
      interpretation,
      constraints,
    );

  const weights =
    dynamicWeights(
      interpretation,
    );

  const calibration =
    evidence
      .objectiveCalibration
      ?.[candidate.id];

  const preference =
    evidence
      .subjectivePreference
      ?.[candidate.id];

  const taskFit =
    taskFitScore(
      candidate,
      interpretation,
    );

  const privacy =
    privacyScore(
      candidate,
      interpretation,
    );

  const latency =
    latencyScore(
      candidate,
      interpretation,
      evidence,
      constraints,
    );

  const cost =
    costScore(
      candidate,
      interpretation,
      constraints,
    );

  const reliability =
    reliabilityScore(
      candidate,
      evidence,
    );

  const components: ModelRouteScoreComponent[] =
    [
      component(
        "objective-calibration",
        calibration
          ?.score ??
        0.5,
        weights[
          "objective-calibration"
        ],
        calibration
          ?.confidence ??
        0.2,
        calibration
          ? "measured"
          : "unknown",
        calibration
          ? [
              `Objective calibration score=${clamp01(calibration.score).toFixed(2)} sampleCount=${calibration.sampleCount ?? "unknown"}.`,
            ]
          : [
              "No objective calibration evidence; neutral score used rather than inventing capability.",
            ],
      ),
      component(
        "task-fit",
        taskFit.score,
        weights[
          "task-fit"
        ],
        0.6,
        "derived",
        taskFit.reasons,
      ),
      component(
        "privacy",
        privacy.score,
        weights.privacy,
        candidate.privacy
          .execution !==
          "unknown"
          ? 0.9
          : 0.4,
        "declared",
        privacy.reasons,
      ),
      component(
        "latency",
        latency.score,
        weights.latency,
        latency.confidence,
        latency.measured
          ? "measured"
          : "unknown",
        latency.reasons,
      ),
      component(
        "cost",
        cost.score,
        weights.cost,
        candidate.cost
          .known ||
        candidate.cost.mode ===
          "local-compute"
          ? 0.9
          : 0.3,
        candidate.cost
          .known ||
        candidate.cost.mode ===
          "local-compute"
          ? "declared"
          : "unknown",
        cost.reasons,
      ),
      component(
        "availability",
        availabilityScore(
          candidate,
        ),
        weights.availability,
        candidate.availability ===
          "unknown"
          ? 0.4
          : 0.95,
        "declared",
        [
          `Registry availability=${candidate.availability}.`,
        ],
      ),
      component(
        "subjective-preference",
        preference
          ?.score ??
        0.5,
        weights[
          "subjective-preference"
        ],
        preference
          ?.confidence ??
        0.2,
        preference
          ? "measured"
          : "unknown",
        preference
          ? [
              `Subjective preference score=${clamp01(preference.score).toFixed(2)} sampleCount=${preference.sampleCount ?? "unknown"}.`,
              "Preference is intentionally separate from objective capability.",
            ]
          : [
              "No subjective preference evidence; neutral score used.",
            ],
      ),
      component(
        "reliability",
        reliability.score,
        weights.reliability,
        reliability.confidence,
        reliability.measured
          ? "measured"
          : "unknown",
        reliability.reasons,
      ),
    ];

  const score =
    exclusions.length >
      0
      ? 0
      : clamp01(
          components.reduce(
            (
              total,
              entry,
            ) =>
              total +
              entry.weightedScore,
            0,
          ),
        );

  const evidenceWeight =
    components.reduce(
      (
        total,
        entry,
      ) =>
        total +
        entry.weight,
      0,
    );

  const confidence =
    exclusions.length >
      0
      ? 1
      : clamp01(
          evidenceWeight >
            0
            ? components.reduce(
                (
                  total,
                  entry,
                ) =>
                  total +
                  entry.confidence *
                    entry.weight,
                0,
              ) /
              evidenceWeight
            : 0,
        );

  const reasons = [
    ...components.flatMap(
      (entry) =>
        entry.reasons.map(
          (reason) =>
            `${entry.component}: ${reason}`,
        ),
    ),
  ];

  return {
    candidate:
      structuredClone(
        candidate,
      ),
    eligible:
      exclusions.length ===
      0,
    exclusionReasons:
      exclusions,
    score,
    confidence,
    components,
    reasons,
  };
}

export class ModelRouter {
  constructor(
    private readonly candidateSource:
      CognitiveModelCandidateSource =
        defaultCandidateSource,
  ) {}

  async route(
    input: {
      interpretation:
        InputInterpretation;
      evidence?:
        ModelRoutingEvidence;
      constraints?:
        ModelRoutingConstraints;
    },
  ): Promise<ModelRouteDecision> {
    const evidence =
      input.evidence ??
      {};

    const constraints =
      structuredClone(
        input.constraints ??
        {},
      );

    const candidates =
      await this.candidateSource();

    const ranking =
      candidates
        .map(
          (candidate) =>
            scoreCandidate(
              candidate,
              input.interpretation,
              evidence,
              constraints,
            ),
        )
        .sort(
          (
            left,
            right,
          ) => {
            if (
              left.eligible !==
              right.eligible
            ) {
              return left.eligible
                ? -1
                : 1;
            }

            if (
              left.score !==
              right.score
            ) {
              return (
                right.score -
                left.score
              );
            }

            if (
              left.confidence !==
              right.confidence
            ) {
              return (
                right.confidence -
                left.confidence
              );
            }

            return left
              .candidate.id
              .localeCompare(
                right
                  .candidate.id,
              );
          },
        );

    const minimum =
      clamp01(
        constraints
          .minimumRouteConfidence ??
        0,
      );

    const selected =
      ranking.find(
        (entry) =>
          entry.eligible &&
          entry.confidence >=
            minimum,
      );

    const warnings:
      string[] =
        [];

    if (
      candidates.length ===
      0
    ) {
      warnings.push(
        "Model registry returned no candidates.",
      );
    }

    if (
      ranking.some(
        (entry) =>
          entry.components.some(
            (componentScore) =>
              componentScore
                .evidence ===
              "unknown",
          ),
      )
    ) {
      warnings.push(
        "One or more routing dimensions lacked measured/declared evidence; neutral conservative scores were used.",
      );
    }

    if (!selected) {
      warnings.push(
        "No eligible model satisfied the routing constraints/confidence threshold.",
      );
    }

    return {
      version:
        MODEL_ROUTER_VERSION,
      routedAt:
        new Date()
          .toISOString(),
      status:
        selected
          ? "selected"
          : "no-eligible-candidate",
      selected:
        selected
          ? structuredClone(
              selected,
            )
          : undefined,
      ranking,
      taskSummary: {
        primaryDomain:
          input
            .interpretation
            .taskIntelligence
            .primaryDomain
            .value,
        complexity:
          input
            .interpretation
            .taskIntelligence
            .complexity.value,
        creativity:
          input
            .interpretation
            .taskIntelligence
            .creativity.value,
        factuality:
          input
            .interpretation
            .taskIntelligence
            .factuality.value,
        reasoningDepth:
          input
            .interpretation
            .taskIntelligence
            .reasoningDepth.value,
        privacySensitivity:
          input
            .interpretation
            .taskIntelligence
            .privacySensitivity
            .value,
        latencyTolerance:
          input
            .interpretation
            .taskIntelligence
            .latencyTolerance
            .value,
        costTolerance:
          input
            .interpretation
            .taskIntelligence
            .costTolerance.value,
        requiredTools: [
          ...(
            input
              .interpretation
              .taskSignals
              .requiredTools ??
            []
          ),
        ],
      },
      constraints,
      warnings,
    };
  }
}

function numericCost(
  candidate:
    ModelRoutingCandidate,
): number {
  if (
    candidate.cost.mode ===
      "local-compute"
  ) {
    return 0;
  }

  if (
    !candidate.cost.known
  ) {
    return Number
      .POSITIVE_INFINITY;
  }

  return (
    candidate.cost
      .inputPerMillionTokens ??
    0
  ) +
  (
    candidate.cost
      .outputPerMillionTokens ??
    0
  );
}

function cheapCandidate(
  route:
    ModelRouteDecision,
): RankedModelCandidate | undefined {
  return eligibleRanking(
    route,
  )
    .slice()
    .sort(
      (
        left,
        right,
      ) => {
        const costDiff =
          numericCost(
            left.candidate,
          ) -
          numericCost(
            right.candidate,
          );

        if (
          Number.isFinite(
            costDiff,
          ) &&
          costDiff !==
          0
        ) {
          return costDiff;
        }

        if (
          left.score !==
          right.score
        ) {
          return (
            right.score -
            left.score
          );
        }

        return left
          .candidate.id
          .localeCompare(
            right
              .candidate.id,
          );
      },
    )[0];
}

function provisionalCompetitionWinner(
  runs:
    readonly MultiModelOrchestrationRun[],
  route:
    ModelRouteDecision,
): MultiModelOrchestrationRun | undefined {
  const successful =
    runs.filter(
      (run) =>
        run.ok &&
        Boolean(
          run.text,
        ),
    );

  return successful
    .slice()
    .sort(
      (
        left,
        right,
      ) => {
        const leftConfidence =
          left.confidence ??
          routeEntryFor(
            route,
            left.candidateId,
          )?.confidence ??
          0;

        const rightConfidence =
          right.confidence ??
          routeEntryFor(
            route,
            right.candidateId,
          )?.confidence ??
          0;

        if (
          leftConfidence !==
          rightConfidence
        ) {
          return (
            rightConfidence -
            leftConfidence
          );
        }

        const leftScore =
          left.routeScore ??
          0;

        const rightScore =
          right.routeScore ??
          0;

        if (
          leftScore !==
          rightScore
        ) {
          return (
            rightScore -
            leftScore
          );
        }

        return left.id
          .localeCompare(
            right.id,
          );
      },
    )[0];
}

            input

            "Refine the prior candidate. Correct weaknesses, improve completeness and clarity, preserve valid content, and do not invent missing evidence.",
            [
              {
                label:
                  "INITIAL CANDIDATE",
                text:
                  draft.text,
              },
            ],
          ),
        );

      if (
        refined.ok &&
        refined.text
      ) {
        return finish(
          "completed",
          refined.text,
          refined.id,
        );
      }

      warnings.push(
        "Refinement pass failed; returning the initial candidate as partial output.",
      );

      return finish(
        "partial",
        draft.text,
        draft.id,
      );
    }

    if (
      strategy ===
        "critic-loop"
    ) {
      const draft =
        await execute(
          primary,
          "primary",
          "Produce the initial candidate for critique.",
          promptForPass(
            input
              .compiledPrompt,
            "Produce the first complete candidate. Preserve constraints, evidence boundaries, assumptions, and uncertainty.",
          ),
        );

      if (
        !draft.ok ||
        !draft.text
      ) {
        return finish(
          "failed",
        );
      }

      const critic =
        await execute(
          primary,
          "critic",
          "Critique the initial candidate against the active task and reasoning contract.",
          promptForPass(
            input
              .compiledPrompt,
            "Critique the prior candidate. Identify concrete factual, logical, evidence, constraint, completeness, and uncertainty defects. Do not rewrite the answer yet.",
            [
              {
                label:
                  "CANDIDATE TO CRITIQUE",
                text:
                  draft.text,
              },
            ],
          ),
        );

      if (
        !critic.ok ||
        !critic.text
      ) {
        warnings.push(
          "Critic pass failed; returning the initial candidate as partial output.",
        );

        return finish(
          "partial",
          draft.text,
          draft.id,
        );
      }

      const refined =
        await execute(
          primary,
          "refinement",
          "Revise the candidate using the explicit critique findings.",
          promptForPass(
            input
              .compiledPrompt,
            "Revise the initial candidate using the critique. Repair supported defects, preserve valid content, and do not invent evidence. Return the revised final artifact.",
            [
              {
                label:
                  "INITIAL CANDIDATE",
                text:
                  draft.text,
              },
              {
                label:
                  "CRITIQUE FINDINGS",
                text:
                  critic.text,
              },
            ],
          ),
        );

      if (
        refined.ok &&
        refined.text
      ) {
        warnings.push(
          "OH-4S completed the strategy-level critic loop. Formal independent verification remains the responsibility of OH-4T.",
        );

        return finish(
          "completed",
          refined.text,
          refined.id,
        );
      }

      warnings.push(
        "Final refinement failed after a successful critic pass; returning the initial candidate as partial output.",
      );

      return finish(
        "partial",
        draft.text,
        draft.id,
      );
    }

    blockedRequirements.push(
      `Unsupported strategy: ${strategy}.`,
    );

    return finish(
      "blocked",
    );
  }
}

function eligibleAlternates(
  modelRoute:
    ModelRouteDecision,
): RankedModelCandidate[] {
  const currentId =
    modelRoute.selected
      ?.candidate.id;

  return modelRoute.ranking
    .filter(
      (entry) =>
        entry.eligible &&
        entry.candidate.id !==
          currentId,
    )
    .sort(
      (
        left,
        right,
      ) => {
        if (
          left.score !==
          right.score
        ) {
          return (
            right.score -
            left.score
          );
        }

        if (
          left.confidence !==
          right.confidence
        ) {
          return (
            right.confidence -
            left.confidence
          );
        }

        return left.candidate.id
          .localeCompare(
            right.candidate.id,
          );
      },
    );
}

  const seen =
    new Set<string>([
      `${selected.kind}:${selected.targetModelId ?? ""}:${selected.nextReasoningEffort ?? ""}`,
    ]);

  const output:
    EscalationDirective[] =
      [];

  for (
    const value
    of values
  ) {
    const key =
      `${value.kind}:${value.targetModelId ?? ""}:${value.nextReasoningEffort ?? ""}`;

    if (
      seen.has(
        key,
      )

      continue;
    }

    seen.add(
      key,

    output.push(
      value,
    );
  }

  return output;

export class EscalationEngine {
  decide(
    input: {
      confidence:
        ConfidenceAssessment;

      verification:
        CriticVerificationResult;

      orchestration:
        MultiModelOrchestrationResult;

      modelRoute:
        ModelRouteDecision;

      reasoningContract:
        ReasoningContract;

      context?:
        EscalationContext;

      policy?:
        Partial<
          EscalationPolicy
        >;
    },

    const policy =
      policyWithOverrides(
        input.policy,
      );

    const context =
      input.context ??
      {};

    const attemptCount =
      nonNegativeInteger(
        context.attemptCount,
        0,
        "attemptCount",
      );

    const maximumAttempts =
      positiveInteger(
        context.maxAttempts,
        policy.maxAttempts,
        "context.maxAttempts",
      );

    const attemptExhausted =
      attemptCount >=
      maximumAttempts;

    const allowRetry =
      context.allowRetry ??
      true;

    const allowContextEnrichment =
      context
        .allowContextEnrichment ??
      true;

    const allowModelSwitch =
      context.allowModelSwitch ??
      true;

    const allowReasoningIncrease =
      context
        .allowReasoningIncrease ??
      true;

    const allowCritic =
      context.allowCritic ??
      true;

    const allowHumanClarification =
      context
        .allowHumanClarification ??
      true;

    const unresolved =
      (
        context
          .unresolvedQuestions ??
        []
      )
        .filter(
          (question) =>
            question.question
              .trim()
              .length >
            0,
        );

    const blockingQuestions =
      unresolved.filter(
        (question) =>
          question.blocking,
      );

    const contextGapCount =
      nonNegativeInteger(
        context.contextGapCount,
        0,
        "contextGapCount",
      );

    const transientFailureCount =
      nonNegativeInteger(
        context.transientFailureCount,
        0,
        "transientFailureCount",
      );

    const modelFailureCount =
      nonNegativeInteger(
        context.modelFailureCount,
        0,
        "modelFailureCount",
      );

    const findingSet =
      findingCodes(
        input.verification,
      );

    const evidenceValue =
      componentValue(
        input.confidence,
        "evidence-quality",
      );

    const evidenceQuality =
      componentQuality(
        input.confidence,
        "evidence-quality",
      );

    const historyValue =
      componentValue(
        input.confidence,
        "model-history",
      );

    const disagreementValue =
      componentValue(
        input.confidence,
        "disagreement",
      );

    const noveltyValue =
      componentValue(
        input.confidence,
        "task-novelty",
      );

    const uncertaintyValue =
      componentValue(
        input.confidence,
        "uncertainty",
      );

    const criticGapDetected =
      (
        input.verification
          .independentCritic
          .required &&
        !input.verification
          .independentCritic
          .parsed
      ) ||
      findingSet.has(
        "independent-critic-executor-missing",
      ) ||
      findingSet.has(
        "independent-critic-candidate-missing",
      ) ||
      findingSet.has(
        "critic-unavailable",
      );

    const contextGapDetected =
      contextGapCount >
        0 ||
      (
        context.contextCanBeEnriched ===
          true &&
        (
          evidenceQuality ===
            "missing" ||
          evidenceQuality ===
            "weak" ||
          (
            evidenceValue !==
              undefined &&
            evidenceValue <
              policy
                .contextEnrichmentThreshold
          ) ||
          findingSet.has(
            "missing-evidence-references",
          ) ||
          findingSet.has(
            "constraint-coverage-missing",
          ) ||
          findingSet.has(
            "factual-coverage-missing",
          )
        )
      );

    const transientFailureDetected =
      transientFailureCount >
        0 ||
      input.orchestration
        .status ===
        "failed" ||
      input.orchestration
        .status ===
        "partial";

    const disagreementHigh =
      disagreementValue !==
        undefined &&
      disagreementValue <
        0.45;

    const historyWeak =
      historyValue !==
        undefined &&
      historyValue <
        0.45;

    const modelUnderperformanceDetected =
      modelFailureCount >
        0 ||
      disagreementHigh ||
      historyWeak;

    const noveltyHigh =
      noveltyValue !==
        undefined &&
      noveltyValue <
        0.45;

    const uncertaintyHigh =
      uncertaintyValue !==
        undefined &&
      uncertaintyValue <
        0.45;

    const currentReasoningEffort =
      context.currentReasoningEffort ??
      "standard";

    const maximumReasoningEffort =
      context.maxReasoningEffort ??
      "maximum";

    const increasedEffort =
      nextReasoningEffort(
        currentReasoningEffort,
        maximumReasoningEffort,
      );

    const reasoningDepthGapDetected =
      Boolean(
        increasedEffort,
      ) &&
      (
        noveltyHigh ||
        uncertaintyHigh ||
        (
          input.confidence.score <
            policy
              .reasoningIncreaseThreshold &&
          input.reasoningContract
            .taskProfile
            .reasoningDepth ===
            "very-deep"
        )
      );

    const blockingAmbiguityDetected =
      context.humanClarificationNeeded ===
        true ||
      blockingQuestions.length >
        0;

    const confidenceSufficient =
      input.confidence.score >=
        policy.acceptanceThreshold &&
      input.verification
        .verdict ===
        "pass" &&
      input.verification
        .trustedConclusion &&
      input.orchestration
        .status ===
        "completed";

    const candidates:
      EscalationDirective[] =
        [];

    if (
      blockingAmbiguityDetected &&
      allowHumanClarification
    ) {
      const questions =
        (
          blockingQuestions.length >
            0
            ? blockingQuestions
            : unresolved
        )
          .slice(
            0,
            policy
              .maxClarificationQuestions,
          )
          .map(
            (question) =>
              question.question
                .trim(),
          );

      candidates.push(
        directive({
          kind:
            "request-human-clarification",
          reasonCode:
            "blocking-ambiguity",
          summary:
            "Request human clarification because unresolved ambiguity blocks reliable continuation.",
          prerequisites: [
            "Present only the unresolved questions needed to continue.",
            "Preserve the current cognitive artifact while waiting for clarification.",
          ],
          preservesCurrentArtifact:
            true,
          clarificationQuestions:
            questions,
        }),
      );
    }

    if (
      contextGapDetected &&
      allowContextEnrichment
    ) {
      candidates.push(
        directive({
          kind:
            "enrich-context",
          reasonCode:
            "context-insufficient",
          summary:
            "Retrieve or assemble additional relevant context/evidence before reasoning again.",
          prerequisites: [
            "Use OH-4K context retrieval rules.",
            "Preserve source provenance and authority boundaries.",
            "Do not treat retrieved context as permission to act.",
          ],
          preservesCurrentArtifact:
            true,
        }),
      );
    }

    if (
      criticGapDetected &&
      allowCritic &&
      context.criticAvailable ===
        true
    ) {
      candidates.push(
        directive({
          kind:
            "invoke-critic",
          reasonCode:
            "verification-critic-needed",
          summary:
            "Run the missing independent critic verification required by the active reasoning contract.",
          prerequisites: [
            "Critic candidate must be independent from the final-source model.",
            "Critic candidate must satisfy privacy and availability gates.",
          ],
          preservesCurrentArtifact:
            true,
          criticCandidateId:
            context
              .criticCandidateId,
        }),
      );
    }

    if (
      transientFailureDetected &&
      allowRetry &&
      !attemptExhausted
    ) {
      candidates.push(
        directive({
          kind:
            "retry",
          reasonCode:
            "transient-failure",
          summary:
            "Retry the bounded cognitive operation because the current failure pattern may be transient.",
          prerequisites: [
            "Retry remains inside the configured attempt budget.",
            "Reuse authoritative inputs unless fresher context is explicitly gathered.",
          ],
          preservesCurrentArtifact:
            true,
          attemptNumber:
            attemptCount +
            1,
        }),
      );
    }

    const alternate =
      eligibleAlternates(
        input.modelRoute,
      )[0];

    if (
      modelUnderperformanceDetected &&
      allowModelSwitch &&
      alternate &&
      input.confidence.score <
        policy.modelSwitchThreshold
    ) {
      candidates.push(
        directive({
          kind:
            "switch-model",
          reasonCode:
            "model-underperforming",
          summary:
            "Switch to the next eligible OH-4M-ranked model because current-model evidence indicates underperformance or disagreement.",
          prerequisites: [
            "Use only an OH-4M eligible alternate candidate.",
            "Re-run the applicable strategy and verification steps after switching.",
          ],
          preservesCurrentArtifact:
            true,
          targetModelId:
            alternate
              .candidate.id,
        }),
      );
    }

    if (
      reasoningDepthGapDetected &&
      allowReasoningIncrease &&
      increasedEffort &&
      input.confidence.score <
        policy
          .reasoningIncreaseThreshold
    ) {
      candidates.push(
        directive({
          kind:
            "increase-reasoning-effort",
          reasonCode:
            "reasoning-depth-insufficient",
          summary:
            "Increase bounded reasoning effort because novelty, uncertainty, or required reasoning depth exceeds the current setting.",
          prerequisites: [
            "Remain inside the active reasoning contract pass budget.",
            "Do not treat increased reasoning effort as authority to act.",
          ],
          preservesCurrentArtifact:
            true,
          nextReasoningEffort:
            increasedEffort,
        }),
      );
    }

    if (
      allowRetry &&
      !attemptExhausted &&
      input.confidence.score <
        policy.retryThreshold
    ) {
      candidates.push(
        directive({
          kind:
            "retry",
          reasonCode:
            "low-confidence-retry",
          summary:
            "Retry the cognitive pass because confidence remains below the bounded retry threshold and no stronger prerequisite necessarily blocks continuation.",
          prerequisites: [
            "Retry remains inside the configured attempt budget.",
            "Re-run downstream verification after retry.",
          ],
          preservesCurrentArtifact:
            true,
          attemptNumber:
            attemptCount +
            1,
        }),
      );
    }

    let selected:
      EscalationDirective;

    if (
      confidenceSufficient
    ) {
      selected =
        directive({
          kind:
            "none",
          reasonCode:
            "confidence-sufficient",
          summary:
            "No escalation is required; confidence and verification satisfy the acceptance threshold.",
          prerequisites:
            [],
          preservesCurrentArtifact:
            true,
        });
    } else if (
      candidates.length >
        0
    ) {
      selected =
        candidates[0];
    } else if (
      attemptExhausted
    ) {
      selected =
        directive({
          kind:
            "stop-safely",
          reasonCode:
            "attempt-budget-exhausted",
          summary:
            "Stop safely because the configured cognitive attempt budget is exhausted and no higher-priority recovery path is available.",
          prerequisites: [
            "Surface the unresolved limitation and preserve the last verified/auditable artifacts.",
          ],
          preservesCurrentArtifact:
            true,
        });
    } else {
      selected =
        directive({
          kind:
            "stop-safely",
          reasonCode:
            "no-safe-recovery-path",
          summary:
            "Stop safely because confidence is insufficient and no permitted recovery path is currently available.",
          prerequisites: [
            "Surface the unresolved limitation instead of fabricating certainty.",
          ],
          preservesCurrentArtifact:
            true,

    }

    }

    return {
      version:
        ESCALATION_ENGINE_VERSION,
      decidedAt:
        new Date()
          .toISOString(),
      required:
        selected.kind !==
        "none",
      confidence: {
        score:
          input.confidence
            .score,
        percentage:
          input.confidence
            .percentage,
        band:
          input.confidence
            .band,
        acceptanceThreshold:
          policy
            .acceptanceThreshold,
      },
      verificationVerdict:
        input.verification
          .verdict,
      selected,
      alternatives,
      attemptBudget: {
        current:
          attemptCount,
        maximum:
          maximumAttempts,
        exhausted:
          attemptExhausted,
      },
      signals: {
        contextGapDetected,
        criticGapDetected,
        transientFailureDetected,
        modelUnderperformanceDetected,
        reasoningDepthGapDetected,
        blockingAmbiguityDetected,
      },
      warnings,
      provenance: {
        confidenceEngineVersion:
          input.confidence
            .version,
        criticVerificationVersion:
          input.verification
            .version,
        orchestratorVersion:
          input.orchestration
            .version,
        modelRouterVersion:
          input.modelRoute
            .version,
        reasoningContractVersion:
          input
            .reasoningContract
            .version,
      },
      invariants: {
        escalationExecuted:
          false,
        toolsExecuted:
          false,
        actionsExecuted:
          false,
        authorityBypassed:
          false,
        modelRouteMutated:
          false,
        verificationMutated:
          false,
      },
    };
  }
}

function shouldUseAuthoritativeAssessmentContext(
  userMessage: string,
  projectId?: string | null
): boolean {
  if (!projectId) {
    return false;
  }

  const normalized = userMessage
    .toLowerCase()
    .replace(/\s+/g, " ")
    .trim();

  const asksForAssessment =
    /\b(assess|assessment|evaluate|evaluation|status|state|health|healthy|attention|facts?|inferences?|predictions?|unknowns?|recommend(?:ed|ation)?s?|actions?)\b/i.test(
      normalized
    );

  const asksForCurrentAuthority =
    /\b(current|active|project|workspace|runtime|world state|evidence|known|scope|scoped)\b/i.test(
      normalized
    );

  return (
    asksForAssessment &&
    asksForCurrentAuthority
  );
}
